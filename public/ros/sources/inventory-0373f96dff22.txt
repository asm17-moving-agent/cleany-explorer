12: class SimulatedEncoderNode(Node):

14: 
15:     Subscriptions:
16:         joint_states (sensor_msgs/msg/JointState): Cumulative simulated angles.
17: 
18:     Publishers:
19:         wheel_encoder/joint_states (sensor_msgs/msg/JointState): Quantized wheel
20:             positions after configured scale and tick noise.
21: 
22:     QoS:
23:         Both directions use sensor_data.
24: 
25:     Notes:
26:         Topic and joint names are configurable defaults. Missing wheel samples
27:         are ignored; synthetic encoder data is not measured hardware feedback.
28:     """

30:     def __init__(self) -> None:

37:         self.declare_parameter('tick_noise_stddev', 0.0)
38:         self.declare_parameter('random_seed', 42)
39:         self.declare_parameter(
40:             'front_left_joint', 'front_left_wheel_joint'
41:         )
42:         self.declare_parameter(
43:             'front_right_joint', 'front_right_wheel_joint'
44:         )
45:         self.declare_parameter('rear_left_joint', 'rear_left_wheel_joint')
46:         self.declare_parameter('rear_right_joint', 'rear_right_wheel_joint')
47:         self.declare_parameter('front_left_scale', 1.0)
48:         self.declare_parameter('front_right_scale', 1.0)
49:         self.declare_parameter('rear_left_scale', 1.0)
50:         self.declare_parameter('rear_right_scale', 1.0)

52:         self._joint_names = (
53:             str(self.get_parameter('front_left_joint').value),
54:             str(self.get_parameter('front_right_joint').value),

59:             ticks_per_revolution=int(

67:             ),
68:             tick_noise_stddev=float(
69:                 self.get_parameter('tick_noise_stddev').value
70:             ),
71:             random_seed=int(self.get_parameter('random_seed').value),

85:     def _on_joint_state(self, message: JointState) -> None:
86:         if len(message.position) < len(message.name):
87:             self.get_logger().warning(
88:                 'Ignoring JointState with fewer positions than names',
89:                 throttle_duration_sec=5.0,
90:             )
91:             return
92: 
93:         position_by_name = dict(zip(message.name, message.position))
94:         missing = [
95:             joint_name
96:             for joint_name in self._joint_names
97:             if joint_name not in position_by_name
98:         ]
99:         if missing:
100:             self.get_logger().warning(
101:                 f'Ignoring JointState missing wheel joints: {missing}',
102:                 throttle_duration_sec=5.0,
103:             )
104:             return
105: 
106:         stamp = message.header.stamp
107:         if stamp.sec == 0 and stamp.nanosec == 0:
108:             stamp = self.get_clock().now().to_msg()
109:         stamp_s = float(stamp.sec) + float(stamp.nanosec) * 1e-9
110:         reading = self._encoder.update(
111:             tuple(
112:                 float(position_by_name[joint_name])
113:                 for joint_name in self._joint_names
114:             ),
115:             stamp_s,
116:         )
117: 
118:         output = JointState()
119:         output.header.stamp = stamp
120:         output.name = list(self._joint_names)
121:         output.position = list(reading.positions_rad)
122:         output.velocity = list(reading.velocities_rad_s)
123:         self._publisher.publish(output)
