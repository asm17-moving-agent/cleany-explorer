87:     left_values = (

164:     def confirm_wrist_target(self, report: dict, step: int) -> dict:
165:         """Confirm the fixed head-camera target from a fresh wrist RGB frame."""
166:         if self.wrist_request is None or not self.wrist_reference_id:
167:             raise RuntimeError('Wrist target handoff is unavailable')
168:         request = deepcopy(self.wrist_request)
169:         request.operation = request.CHECK

172:         if (response.reference_id != self.wrist_reference_id
173:                 or response.source_snapshot_id != request.source_snapshot_id
174:                 or response.source_object_id != request.source_object_id):

299:     @timed_operation('reverse_pickup')
300:     def reverse_to_pickup(self, arm: str, target: Pose, policy: PickupReturnPolicy) -> dict:
301:         paths = getattr(self, '_pickup_departure_paths', None)
302:         if not paths or getattr(self, '_pickup_departure_arm', None) != arm:
303:             raise CommandPathValidationError('Reverse return has no recorded departure for this arm')
304:         # Consume once; a failed replay may have moved with the payload attached.
305:         self._pickup_departure_paths = self._pickup_departure_arm = None
306:         validate_command_history(paths, ARM_JOINT_NAMES[arm])
307:         state = self.real.wait_for_settle(ARM_JOINT_NAMES[arm])
308:         hold = self.real.arm_holds.get(arm)
309:         if hold is None or any(abs(a-b) > 1e-9 for a, b in zip(
310:                 hold.command, paths[-1].points[-1].positions, strict=True)):
311:             raise CommandPathValidationError('Reverse return history no longer matches the held command')
312:         goal = self.cartesian_pose(target)
313:         self.motion.set_current_state(state)
314:         def fk(q):
315:             self.real.monitor()
316:             value = self.motion.forward_kinematics(arm, JointSolution(ARM_JOINT_NAMES[arm], q))
317:             return CartesianPose(value.position, value.orientation_xyzw)
318:         positions = dict(zip(state.name, state.position, strict=True))
319:         measured_joints = tuple(positions[name] for name in ARM_JOINT_NAMES[arm])
320:         actual = fk(measured_joints)
321:         commanded = fk(tuple(hold.command))
322:         position_tolerance = self.real.value('hardware_tcp_position_tolerance_m')
323:         orientation_tolerance = math.radians(self.real.value('hardware_tcp_orientation_tolerance_deg'))
324:         # The next movement starts at measured FK. The saved motor commands
325:         # supply relative displacement while retaining loaded-arm support.
326:         # Height correction deliberately makes their absolute TCP differ from
327:         # the physical goal; that difference is not an arrival failure here.
328:         residual = tuple(c-a for c, a in zip(commanded.position, actual.position, strict=True))
329:         reverse = reversed_command_history(paths)
330:         stop_ns = self.pickup_return_height_time(reverse,
331:             lambda q: actual.position[2]+fk(q).position[2]-commanded.position[2],
332:             goal.position[2])
333:         endpoint = fk(reverse.sample_at(stop_ns).positions)
334:         predicted = estimate_measured_path_pose(actual, commanded, endpoint)
335:         validate_pose_endpoint(predicted, goal, position_tolerance, orientation_tolerance)
336:         details = dict(return_route='reversed_departure',
337:                        return_start_source='measured',
338:                        return_measured_start_joint_rad=measured_joints,
339:                        return_measured_start_position_m=actual.position,
340:                        return_measured_start_orientation_xyzw=actual.orientation,
341:                        return_collision_checks_enabled=policy.check_collisions,
342:                        return_recorded_trajectories=len(paths),
343:                        return_source_duration_sec=stop_ns/1e9,
344:                        return_tracking_residual_m=residual,
345:                        return_command_endpoint_m=endpoint.position)
346:         self.event('return_to_pickup_replay', arm=arm, **details)
347:         cursor = 0
348:         corrections = []
349:         details['return_height_corrections'] = corrections
350:         for attempt in range(policy.correction_max_attempts+1):
351:             self.real.execute_return_segment(arm, reverse, cursor, stop_ns,
352:                                              check_collisions=policy.check_collisions,
353:                                              endpoint_pose=goal)
354:             cursor = stop_ns
355:             state = self.real.wait_for_settle(ARM_JOINT_NAMES[arm])
356:             actual = self.cartesian_pose(self.pose(arm))
357:             validate_pose_endpoint(actual, goal, position_tolerance, orientation_tolerance)
358:             error = actual.position[2]-goal.position[2]
359:             sample = dict(attempt=attempt, target_z_m=goal.position[2],
360:                           actual_z_m=actual.position[2], height_error_m=error)
361:             corrections.append(sample)
362:             self.event('pickup_return_height_checked', arm=arm, **sample)
363:             if abs(error) <= policy.height_tolerance_m+1e-9:
364:                 break
365:             # A high arrival may continue down the saved path in small bounded
366:             # steps. Low arrivals, unrelated pose errors and faults stop here.
367:             if error < 0 or attempt == policy.correction_max_attempts:
368:                 raise ValueError('Measured TCP did not reach the raised pickup release height: '
369:                                  f'target={goal.position[2]:.6f}m actual={actual.position[2]:.6f}m '
370:                                  f'error={error:.6f}m tolerance={policy.height_tolerance_m:.6f}m')
371:             self.motion.set_current_state(state)
372:             command = reverse.sample_at(cursor).positions
373:             hold = self.real.arm_holds.get(arm)
374:             if hold is None or any(abs(a-b) > 1e-9 for a,b in zip(command,hold.command,strict=True)):
375:                 raise CommandPathValidationError('Return height correction lost its held command')
376:             command_pose = fk(command)
377:             step = min(error, policy.correction_max_step_m)
378:             stop_ns = self.pickup_return_height_time(
379:                 reverse, lambda q: fk(q).position[2], command_pose.position[2]-step,
380:                 start_ns=cursor)

383:             validate_pose_endpoint(predicted, goal, position_tolerance, orientation_tolerance)
384:             self.event('pickup_return_height_correction', arm=arm, attempt=attempt+1,
385:                        step_m=step, actual_z_m=actual.position[2], target_z_m=goal.position[2])
386:         return details

405:     def pickup_release_retreat(self, arm: str, pregrasp: Pose) -> None:

432:     def check_pickup_arrival(self, arm: str) -> None:
433:         """Confirm the configured release destination using fresh measured state."""
434:         policy = PickupReturnPolicy.from_config(self.config)
435:         if policy.height_offset_m > 0:
436:             target = self.cartesian_pose(self.pickup_destination_pose(arm))
437:             self.real.monitor()
438:             actual = self.cartesian_pose(self.pose(arm))
439:             validate_pose_endpoint(actual, target,
440:                                    self.real.value('hardware_tcp_position_tolerance_m'),
441:                                    math.radians(self.real.value('hardware_tcp_orientation_tolerance_deg')))
442:             # The general 30mm pose envelope alone would accept the old height
443:             # for a +20mm goal. Keep release strictly above the pickup height.
444:             height_error = actual.position[2] - target.position[2]

563:     @staticmethod
564:     def retarget_wrist_roll(trajectory: JointTrajectory, measured,
565:                             arm: str, target_roll: float | None) -> None:
566:         """Keep position-only Cartesian IK from choosing an arbitrary tool roll."""
567:         wrist = f'{arm}_wrist_roll_joint'
568:         if wrist not in trajectory.joint_names:
569:             raise RuntimeError('Cartesian response is missing wrist roll')
570:         positions = dict(zip(measured.name, measured.position))
571:         if wrist not in positions or not math.isfinite(positions[wrist]):
572:             raise RuntimeError('Measured wrist roll is unavailable')
573:         start = float(positions[wrist])
574:         target_roll = start if target_roll is None else float(target_roll)
575:         if not math.isfinite(target_roll):
576:             raise RuntimeError('Cartesian wrist-roll target is not finite')
577:         if len(trajectory.points) < 2:
578:             raise RuntimeError('Cartesian response requires at least two points')
579:         index = trajectory.joint_names.index(wrist)
580:         denominator = len(trajectory.points)-1
581:         for number, point in enumerate(trajectory.points):
582:             values = list(point.positions)
583:             if len(values) != len(trajectory.joint_names):
584:                 raise RuntimeError('Cartesian response point width mismatch')
585:             fraction = number/denominator
586:             values[index] = start+(target_roll-start)*fraction
587:             point.positions = values

695:     def correct_height_endpoint(self, arm: str, target: Pose, phase: str,
696:                                 report: dict | None = None, *,
697:                                 check_collisions: bool = True) -> None:
698:         """Correct lift shortfall beyond tolerance once, without a final height gate."""
699:         policy = HeightCorrectionPolicy.from_config(self.config)
700:         if not policy.enabled:
701:             return
702:         if phase != 'lift':
703:             raise ValueError('Height correction is restricted to lift')
704:         goal = self.cartesian_pose(deepcopy(target))
705:         position_tolerance = self.real.value('hardware_tcp_position_tolerance_m')
706:         orientation_tolerance = math.radians(self.real.value('hardware_tcp_orientation_tolerance_deg'))
707:         history = []
708:         summary = {'target_z_m': goal.position[2], 'tolerance_m': policy.tolerance_m,
709:                    'history': history, 'status': 'checking'}
710:         if report is not None:
711:             report.setdefault('height_corrections', {})[phase] = summary
712:         try:
713:             # Includes fresh feedback, fault, cancellation and payload gates.
714:             state = self.real.wait_for_settle(ARM_JOINT_NAMES[arm])
715:             actual = self.cartesian_pose(self.pose(arm))
716:             validate_height_correctable_endpoint(actual, goal, position_tolerance,
717:                 orientation_tolerance)
718:             hold = self.real.arm_holds.get(arm)
719:             if hold is None:
720:                 raise RuntimeError('Height correction requires a completed held arm command')
721:             self.motion.set_current_state(state)
722:             fk = self.motion.forward_kinematics(
723:                 arm, JointSolution(ARM_JOINT_NAMES[arm], hold.command))
724:             # Retain the sent command's support against the tracking residual.
725:             command = CartesianPose(fk.position, fk.orientation_xyzw)
726:             correction = HeightCorrection(goal.position[2], command.position[2], policy)
727:             summary['initial_command_z_m'] = command.position[2]
728:             sample = {'attempt': 0, 'target_z_m': goal.position[2],
729:                       'command_z_m': command.position[2], 'actual_z_m': actual.position[2],
730:                       'error_m': goal.position[2] - actual.position[2],
731:                       'total_correction_m': 0.}
732:             history.append(sample)
733:             summary.update(sample)
734:             self.event('height_correction_measured', arm=arm, phase=phase, **sample)
735:             next_z = correction.next_command(actual.position[2])
736:             if next_z is None:
737:                 summary['status'] = 'not_needed'
738:             else:
739:                 command_waypoint = CartesianPose(
740:                     (command.position[0], command.position[1], next_z), command.orientation)
741:                 step = next_z-command.position[2]
742:                 # Physical planning starts at measured FK; motor support remains
743:                 # separate from the requested physical upward increment.
744:                 waypoint = self.pose_message(CartesianPose(
745:                     (*goal.position[:2], actual.position[2]+step), goal.orientation))
746:                 self.event('height_correction_command', arm=arm, phase=phase,
747:                            attempt=1, target_z_m=goal.position[2], command_z_m=next_z,
748:                            actual_z_m=actual.position[2], correction_step_m=step,
749:                            collision_checks_enabled=check_collisions)
750:                 self.linear(arm, waypoint, phase+'_height_correction',
751:                             check_orientation=True, check_collisions=check_collisions,
752:                             command_target=command_waypoint, validate_command_path=True)
753:                 # Record the remaining height error without retrying or gating return.
754:                 final = self.cartesian_pose(self.pose(arm))
755:                 sample = {'attempt': 1, 'target_z_m': goal.position[2],
756:                           'command_z_m': next_z, 'actual_z_m': final.position[2],
757:                           'error_m': goal.position[2]-final.position[2],
758:                           'total_correction_m': step}
759:                 history.append(sample)
760:                 summary.update(sample, status='applied')
761:             self.event('height_correction_completed', arm=arm, phase=phase, **summary)
762:         except Exception as error:
763:             summary.update(status='failed', message=str(error))
764:             self.event('height_correction_failed', arm=arm, phase=phase, **summary)
765:             raise

767:     @timed_operation('cartesian_planning')
768:     def _cartesian_trajectory(self, arm: str, measured: JointState,
769:                               waypoints: tuple[CartesianPose, ...], stage: str,
770:                               wrist_roll_target: float | None,
771:                               check_collisions: bool) -> JointTrajectory:
772:         """Plan in the explicitly supplied physical or motor-command frame."""
773:         request = GetCartesianPath.Request()
774:         request.header.frame_id = 'base_link'
775:         request.group_name = f'{arm}_grasp_arm'
776:         request.link_name = f'{arm}_grasp_tcp'
777:         request.start_state.joint_state = deepcopy(measured)
778:         request.start_state.is_diff = True  # retain payload in planning state
779:         request.waypoints = [self.pose_message(point) for point in waypoints]
780:         request.max_step = self.config['cartesian_step_m']
781:         request.jump_threshold = 2.
782:         request.revolute_jump_threshold = .15
783:         request.avoid_collisions = check_collisions
784:         response = self.call(self.cartesian, request, 30.)
785:         if not math.isfinite(response.fraction) or not 0 <= response.fraction <= 1:
786:             raise RuntimeError(f'{stage}: invalid Cartesian path fraction')
787:         if response.error_code.val != 1 or response.fraction < 1.-1e-6:
788:             if response.error_code.val in (
789:                     MoveItErrorCodes.SUCCESS, MoveItErrorCodes.PLANNING_FAILED,
790:                     MoveItErrorCodes.START_STATE_IN_COLLISION, MoveItErrorCodes.GOAL_IN_COLLISION,
791:                     MoveItErrorCodes.NO_IK_SOLUTION):
792:                 raise CartesianPathUnavailable(
793:                     f'{stage}: complete Cartesian path unavailable: '
794:                     f'code={response.error_code.val} fraction={response.fraction:.6f}',
795:                     error_code=response.error_code.val, fraction=response.fraction)
796:             raise RuntimeError(f'{stage}: complete Cartesian path unavailable')
797:         trajectory = response.solution.joint_trajectory
798:         if tuple(trajectory.joint_names) != ARM_JOINT_NAMES[arm]:
799:             raise RuntimeError('Cartesian response joint order mismatch')
800:         # KDL is deliberately position-only for this five-axis arm.  The
801:         # unconstrained wrist roll does not move the TCP position, so the
802:         # Cartesian service may otherwise replace it with an arbitrary value.
803:         # Restore a continuous roll ending at the already validated grasp roll
804:         # (or hold the current roll for retreat/lift) before any validation or
805:         # command is allowed.
806:         CollectionPipeline.retarget_wrist_roll(
807:             trajectory, request.start_state.joint_state, arm, wrist_roll_target)
808:         return trajectory

810:     def linear(self, arm: str, target: Pose, stage: str,
811:                wrist_roll_target: float | None = None, *,
812:                check_orientation: bool = True, route: ApproachRoute | None = None,
813:                check_collisions: bool = True, validate_command_path: bool = False,
814:                command_target: CartesianPose | None = None):
815:         """Plan physical geometry from measured FK, then convert motor support."""
816:         if type(check_collisions) is not bool:
817:             raise ValueError('check_collisions must be boolean')
818:         if command_target is not None and (stage != 'lift_height_correction'
819:                                           or not isinstance(command_target, CartesianPose)):
820:             raise ValueError('Explicit command target is restricted to lift height correction')
821:         defer_shortfall = (stage in ('retreat', 'lift', 'lift_height_correction')
822:                            and height_correction_enabled(self.config))
823:         measured = self.real.monitor()
824:         self.motion.set_current_state(measured)
825:         names = ARM_JOINT_NAMES[arm]
826:         positions = dict(zip(measured.name, measured.position, strict=True))
827:         measured_joints = tuple(positions[name] for name in names)
828:         def fk(q):
829:             self.real.monitor()
830:             value = self.motion.forward_kinematics(arm, JointSolution(names, tuple(q)))
831:             return CartesianPose(value.position, value.orientation_xyzw)
832:         start, goal = fk(measured_joints), self.cartesian_pose(target)
833:         self.event(stage+'_measured_start', arm=arm, start_source='measured',
834:                    measured_joint_rad=list(measured_joints), measured_tcp_position_m=list(start.position))
835:         if route is not None and route.waypoints[-1] != goal:
836:             raise RuntimeError('Approach detour changed the approved grasp target')
837:         waypoints = (goal,) if route is None else route.waypoints
838:         def validate(poses, origin):
839:             if route is None:
840:                 validate_corridor_samples(
841:                     poses, origin, goal,
842:                     self.config['cartesian_tolerance_m'],
843:                     math.radians(15) if check_orientation else None,
844:                     math.radians(30) if check_orientation else None)
845:             else:
846:                 validate_detour_samples(
847:                     poses, origin, route, radius_m=self.config['cartesian_tolerance_m'],
848:                     maximum_extra_length_m=self.config['approach_collision_max_extra_length_m'])
849: 
850:         # This request always starts from one measured snapshot and uses the
851:         # unchanged physical destination. Previous command FK cannot replace it.
852:         trajectory = self._cartesian_trajectory(
853:             arm, measured, waypoints, stage, wrist_roll_target, check_collisions)
854:         try:
855:             nominal = executable_joint_path([tuple(p.positions) for p in trajectory.points])
856:             nominal_poses = []
857:             for index, sample in enumerate(nominal.samples(.01, 2, 10000)):
858:                 if check_collisions and not self.motion.state_is_valid(
859:                         arm, JointSolution(names, sample.positions)):
860:                     raise CollisionValidationError(
861:                         f'{stage}: wrist-roll-preserved Cartesian path is in collision: sample={index}',
862:                         phase='cartesian_path', sample_index=index)
863:                 nominal_poses.append(fk(sample.positions))
864:             validate(nominal_poses, start)
865:         except ValueError as error:
866:             raise CartesianPathUnavailable(
867:                 f'{stage}: Cartesian FK corridor rejected: {error}',
868:                 error_code=MoveItErrorCodes.PLANNING_FAILED, fraction=1.) from error
869: 
870:         # Motor support is an explicit conversion after measured geometry has
871:         # been checked. Its FK is a command coordinate, never the current TCP.
872:         # Opening the jaw does not cancel the arm's active position command.
873:         # Keep its support at release-to-retreat handoff; physical waypoints
874:         # still originate at measured FK and retain the requested destination.
875:         hold = (None if stage == 'return_to_pickup' else
876:                 self.real.arm_holds.get(arm))
877:         command_start = None
878:         if hold is not None:
879:             command_start = fk(hold.command)

1381:     def confirm_visual_observation(
1382:         self,
1383:         reference_id: str,
1384:         observation,
1385:         reference_center: tuple[float, float, float],
1386:         previous_center: tuple[float, float, float],
1387:         previous_observation,
1388:     ) -> dict:
1389:         """Median-filter a suspicious surface centroid before it can stop motion."""
1390:         if observation is None:
1391:             return {'accepted': False, 'suspicious': True, 'observation': None,
1392:                     'reason': 'head_target_mask_or_depth_unavailable'}
1393:         soft_limit = float(self.config['visual_servo_maximum_correction_m'])
1394:         hard_limit = float(self.config.get(
1395:             'visual_servo_hard_correction_limit_m', 2.*soft_limit))
1396:         increment_limit = float(self.config.get(
1397:             'visual_observation_increment_limit_m', .01))
1398:         extent_limit = float(self.config.get(
1399:             'visual_observation_extent_ratio_limit', 1.5))
1400:         consistency = float(self.config.get(
1401:             'visual_observation_consistency_m', .008))
1402:         confirmation_frames = int(self.config.get(
1403:             'visual_observation_confirmation_frames', 3))
1404:         raw_center = observation_center(observation)
1405:         raw_correction = math.dist(reference_center, raw_center)
1406:         raw_increment = math.dist(previous_center, raw_center)
1407:         extent_ratio = observation_extent_ratio(
1408:             previous_observation, observation)
1409:         suspicious = (
1410:             raw_correction > soft_limit
1411:             or raw_increment > increment_limit
1412:             or extent_ratio > extent_limit
1413:         )
1414:         observations = [observation]
1415:         if suspicious:
1416:             for _ in range(confirmation_frames-1):
1417:                 capture = CollectionPipeline.observe_for_approach(
1418:                     self, reference_id, observation_stamp_ns(observations[-1]))
1419:                 if capture is None:
1420:                     return {'accepted': False, 'suspicious': True, 'observation': None,
1421:                             'reason': 'head_target_lost_during_confirmation'}
1422:                 observations.append(capture)
1423:         centers = [observation_center(item) for item in observations]
1424:         filtered_center = coordinate_median(centers)
1425:         inliers = sum(
1426:             math.dist(center, filtered_center) <= consistency
1427:             for center in centers
1428:         )
1429:         correction = math.dist(reference_center, filtered_center)
1430:         increment = math.dist(previous_center, filtered_center)
1431:         consistent = inliers >= len(centers)//2+1
1432:         accepted = (
1433:             consistent
1434:             and extent_ratio <= extent_limit
1435:             and correction <= hard_limit
1436:             and (correction <= soft_limit or increment <= increment_limit)
1437:         )
1438:         return {
1439:             'accepted': accepted,
1440:             'suspicious': suspicious,
1441:             'observation': observations[-1],
1442:             'confirmation_frames': len(observations),
1443:             'confirmation_inliers': inliers,
1444:             'raw_center_m': raw_center,
1445:             'filtered_center_m': filtered_center,
1446:             'raw_correction_m': raw_correction,
1447:             'filtered_correction_m': correction,
1448:             'filtered_increment_m': increment,
1449:             'extent_ratio': extent_ratio,
1450:             'soft_limit_m': soft_limit,
1451:             'hard_limit_m': hard_limit,
1452:             'stable_capture_stamp_ns': min(
1453:                 observation_stamp_ns(item) for item in observations
1454:                 if math.dist(observation_center(item), filtered_center) <= consistency
1455:             ) if accepted else None,
1456:         }

2015:         report = {'success':False,'cycles':[],'empty_confirmations':0}
2016:         last_capture = 0
2017:         try:
2018:             for _ in range(int(self.config.get('maximum_cycles',20))):
2019:                 cycle = self.run(query)
2020:                 report['cycles'].append(cycle)
2021:                 if cycle['success']:
2022:                     if temporary_return_mode(self.config):
2023:                         report.update(success=True, message='Single temporary pick/return/release cycle completed; placement assumed')
2024:                         break
2025:                     if hold_mode(self.config):
2026:                         report.update(success=True,
2027:                                       message='Single grasp/lift/hold cycle completed')
2028:                         break
2029:                     report['empty_confirmations'] = 0
2030:                     continue
2031:                 # No eligible target is NOT automatically an empty workspace.
2032:                 if (cycle.get('error_code') == 2 and cycle.get('detection_count') == 0
2033:                         and 'cleanup_error' not in cycle):
2034:                     self.real.monitor()
2035:                     capture = cycle.get('capture_stamp_ns',0)
2036:                     if capture <= last_capture:
2037:                         raise RuntimeError('Empty observation did not advance')
2038:                     last_capture = capture
2039:                     report['empty_confirmations'] += 1
2040:                     if report['empty_confirmations'] >= self.config['empty_confirmations']:
2041:                         report.update(success=True,message='Consecutive empty observations confirmed')
2042:                         break
2043:                     continue
2044:                 raise RuntimeError(cycle.get('message','Collection failed'))
2045:             else:
2046:                 raise RuntimeError('Cycle limit reached without confirmed empty workspace')
2047:         except BaseException as error:
2048:             report['message'] = str(error)
2049:             report['stop'] = self.real.fail_stop()
2050:         report['events'] = self.events
2051:         report['payload_attached_or_uncertain'] = self.carrying is not None
