22:         path = Path(raw).expanduser()
23:         return path if path.is_absolute() else repository_root / path
24:     config = json.loads(config_file.read_text())
25:     if config['arm'] not in ('left', 'right') or config['frame_id'] != config['arm'] + '_wrist_rgb_optical_frame':

27:     camera = json.loads(resolve(config['camera_config']).read_text())
28:     calibration = json.loads(resolve(config['calibration_config']).read_text())
29:     path = resolve(calibration['intrinsics_file'])

33:         if source[key] != camera[key]:
34:             raise ValueError('Wrist calibration mismatch: ' + key)

36:         raise ValueError('Wrist calibration image dimensions mismatch')

43: class WristRgbCamera(Node):
44:     def __init__(self, **kwargs):
45:         super().__init__('wrist_rgb_camera', **kwargs)
46:         self.declare_parameter('repository_root', '')
47:         self.declare_parameter('config_file', '')
48:         self.declare_parameter('start_enabled', False)
49:         self.declare_parameter('preview_rate_hz', 5.0)
50:         self.declare_parameter('preview_jpeg_quality', 75)
51:         self.config, intrinsic = load_camera_profile(
52:             Path(self.get_parameter('config_file').value),
53:             Path(self.get_parameter('repository_root').value))
54:         width, height = self.config['width'], self.config['height']
55:         self.preview_rate = float(self.get_parameter('preview_rate_hz').value)
56:         self.preview_quality = self.get_parameter('preview_jpeg_quality').value
57:         if not math.isfinite(self.preview_rate) or not 0 < self.preview_rate <= 30:
58:             raise ValueError('Preview rate must be in (0, 30] Hz')
59:         self.preview_rate = min(self.preview_rate, self.config['publish_rate_hz'])
60:         if type(self.preview_quality) is not int or not 1 <= self.preview_quality <= 100:
61:             raise ValueError('Preview JPEG quality must be an integer in [1, 100]')
62:         self.next_preview_at = 0.0
63:         matrix = np.asarray(intrinsic['K'], dtype=float).reshape(3, 3)
64:         distortion = np.asarray(intrinsic['D'], dtype=float).ravel()
65:         if (not np.isfinite(matrix).all() or not np.isfinite(distortion).all()
66:                 or matrix[0, 0] <= 0 or matrix[1, 1] <= 0):
67:             raise ValueError('Invalid wrist camera intrinsics')
68:         self.maps = cv2.initUndistortRectifyMap(
69:             matrix, distortion, None, matrix, (width, height), cv2.CV_32FC1)
70:         self.info = CameraInfo(width=width, height=height, distortion_model='plumb_bob',
71:             d=[0.] * 5, k=matrix.ravel().tolist(), r=np.eye(3).ravel().tolist(),
72:             p=np.column_stack((matrix, np.zeros(3))).ravel().tolist())
73:         topic = '/' + self.config['arm'] + '_wrist_camera'
74:         self.images = self.create_publisher(Image, topic + '/image_raw', qos_profile_sensor_data)
75:         self.infos = self.create_publisher(CameraInfo, topic + '/camera_info', qos_profile_sensor_data)
76:         self.previews = self.create_publisher(
77:             CompressedImage, topic + '/preview/compressed', qos_profile_sensor_data)
78:         self.camera = None
79:         self.enabled_state = self.create_publisher(Bool, topic + '/enabled', QoSProfile(
80:             depth=1, reliability=ReliabilityPolicy.RELIABLE,
81:             durability=DurabilityPolicy.TRANSIENT_LOCAL))
82:         self.state_timer = self.create_timer(1.0, self.publish_enabled)
83:         self.publish_enabled()
84:         self.enable_service = self.create_service(SetBool, topic + '/set_enabled', self.set_enabled)
85:         self.timer = self.create_timer(1. / self.config['publish_rate_hz'], self.publish_frame)
86:         if self.get_parameter('start_enabled').value:

89:     def open_camera(self) -> None:
90:         if self.camera is not None:
91:             return
92:         camera = cv2.VideoCapture(self.config['camera_device'], cv2.CAP_V4L2)
93:         try:
94:             if not camera.isOpened():
95:                 raise RuntimeError('Cannot open configured wrist USB camera')
96:             camera.set(cv2.CAP_PROP_FRAME_WIDTH, self.config['width'])
97:             camera.set(cv2.CAP_PROP_FRAME_HEIGHT, self.config['height'])
98:             camera.set(cv2.CAP_PROP_FPS, self.config['publish_rate_hz'])
99:             camera.set(cv2.CAP_PROP_BUFFERSIZE, 1)
100:         except Exception:
101:             camera.release()
102:             raise
103:         self.camera = camera
104:         self.publish_enabled()
105:         self.get_logger().info(f"Wrist camera enabled: {self.config['arm']}")

115:     def set_enabled(self, request: SetBool.Request, response: SetBool.Response) -> SetBool.Response:
116:         try:
117:             if request.data:

119:             else:

121:         except Exception as error:
122:             response.success, response.message = False, str(error)
123:         else:

134:     def publish_frame(self) -> None:
135:         if self.camera is None:
136:             return
137:         # USB driver supplies no exposure timestamp here: stamp host acquisition.
138:         ok, frame = self.camera.read()
139:         expected = (self.config['height'], self.config['width'], 3)
140:         if not ok or frame.shape != expected:
141:             self.get_logger().error('Missing wrist RGB or unexpected image dimensions')
142:             return
143:         stamp = self.get_clock().now().to_msg()
144:         rectified = cv2.remap(frame, *self.maps, interpolation=cv2.INTER_LINEAR)
145:         message = Image(height=expected[0], width=expected[1], encoding='bgr8',
146:                         step=expected[1] * 3, data=rectified.tobytes())
147:         message.header.stamp, message.header.frame_id = stamp, self.config['frame_id']
148:         info = deepcopy(self.info)
149:         info.header = deepcopy(message.header)
150:         self.images.publish(message)
151:         self.infos.publish(info)
152:         now = time.monotonic()
153:         if self.previews.get_subscription_count() and now >= self.next_preview_at:
154:             period = 1. / self.preview_rate
155:             if not self.next_preview_at or now - self.next_preview_at >= period:
156:                 self.next_preview_at = now + period
157:             else:
158:                 self.next_preview_at += period
159:             encoded, jpeg = cv2.imencode(
160:                 '.jpg', rectified, [cv2.IMWRITE_JPEG_QUALITY, self.preview_quality])
161:             if encoded:
162:                 preview = CompressedImage(
163:                     header=deepcopy(message.header), format='bgr8; jpeg compressed bgr8',
164:                     data=jpeg.tobytes())
165:                 self.previews.publish(preview)
166:             else:
167:                 self.get_logger().error('Wrist preview JPEG encoding failed')
