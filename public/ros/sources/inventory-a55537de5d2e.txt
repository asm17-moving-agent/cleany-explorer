15: def quaternion(matrix: np.ndarray) -> tuple[float, float, float, float]:
16:     cr, cp, cy = np.cos(np.array(rpy(matrix))/2)
17:     sr, sp, sy = np.sin(np.array(rpy(matrix))/2)
18:     return (sr*cp*cy-cr*sp*sy, cr*sp*cy+sr*cp*sy,
19:             cr*cp*sy-sr*sp*cy, cr*cp*cy+sr*sp*sy)

22: class GazeboPanTfPublisher(Node):
23:     """Publish the simulated camera mount transform from measured head pan.
24: 
25:     Subscriptions:
26:         joint_states (sensor_msgs/msg/JointState): head_pan_joint angle in radians.
27: 
28:     TF:
29:         Broadcasts base_link -> head_camera_depth_frame at the joint timestamp,
30:         rotating configured mount geometry about pan_pivot.
31: 
32:     QoS:
33:         Joint input uses sensor_data.
34: 
35:     Notes:
36:         Missing or non-finite pan positions are ignored. Use only when this node
37:         owns the camera transform instead of a robot-state publisher.
38:     """
39:     def __init__(self) -> None:
40:         super().__init__('gazebo_pan_tf_publisher')
41:         for name, value in dict(translation=[0., 0., 0.], rotation_rpy=[0., 0., 0.],
42:                                pan_pivot=[0., 0., 0.]).items():
43:             self.declare_parameter(name, value)
44:         self.camera = np.array(self.get_parameter('translation').value)
45:         self.pivot = np.array(self.get_parameter('pan_pivot').value)
46:         self.rotation = rotation(self.get_parameter('rotation_rpy').value)
47:         self.tf = TransformBroadcaster(self)
48:         self.create_subscription(JointState, 'joint_states', self.on_joint, qos_profile_sensor_data)
49: 
50:     def on_joint(self, msg: JointState) -> None:
51:         if 'head_pan_joint' not in msg.name:
52:             return
53:         index = msg.name.index('head_pan_joint')
54:         if index >= len(msg.position) or not np.isfinite(msg.position[index]):
55:             return
56:         turn = rotation([0., 0., msg.position[index]])
57:         xyz = self.pivot+turn@(self.camera-self.pivot)
58:         q = quaternion(turn@self.rotation)
59:         tf = TransformStamped()
60:         tf.header.stamp = msg.header.stamp
61:         tf.header.frame_id, tf.child_frame_id = 'base_link', 'head_camera_depth_frame'
62:         tf.transform.translation.x, tf.transform.translation.y, tf.transform.translation.z = map(float, xyz)
63:         tf.transform.rotation.x, tf.transform.rotation.y, tf.transform.rotation.z, tf.transform.rotation.w = map(float, q)
64:         self.tf.sendTransform(tf)
