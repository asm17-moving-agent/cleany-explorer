310:     @staticmethod
311:     def _validate_candidates(candidates, planning_frame: str) -> None:
312:         if not candidates:
313:             raise ValueError('at least one candidate is required')
314:         if not planning_frame:
315:             raise ValueError('planning_frame must not be empty')
316:         first = candidates[0]
317:         if not first.snapshot_id or first.object_id == 0 or not first.header.frame_id:
318:             raise ValueError('candidate snapshot, object, and frame are required')
319:         if first.target_object.object_id != first.object_id:
320:             raise ValueError('target OBB object ID does not match candidate')
321:         if first.header.frame_id != planning_frame:
322:             raise ValueError(
323:                 'candidate frame must match configured planning_frame '
324:                 f'{planning_frame!r}'
325:             )
326:         size = first.target_object.obb_size
327:         pose = first.target_object.obb_pose
328:         values = (
329:             size.x, size.y, size.z,
330:             pose.position.x, pose.position.y, pose.position.z,
331:             pose.orientation.x, pose.orientation.y, pose.orientation.z, pose.orientation.w,
332:         )
333:         if not all(math.isfinite(value) for value in values) or min(size.x, size.y, size.z) <= 0:
334:             raise ValueError('target OBB must have finite positive dimensions and pose')
335:         norm = math.sqrt(sum(value * value for value in values[-4:]))
336:         if not math.isclose(norm, 1.0, abs_tol=1e-4):
337:             raise ValueError('target OBB quaternion must be normalized')
338:         for candidate in candidates:
339:             if (
340:                 candidate.snapshot_id != first.snapshot_id
341:                 or candidate.object_id != first.object_id
342:                 or candidate.header.frame_id != first.header.frame_id
343:                 or candidate.target_object != first.target_object
344:             ):
345:                 raise ValueError('all candidates must share snapshot, object, frame, and OBB')

347:     def _current_joint_state(self) -> JointState:
348:         with self._joint_state_lock:
349:             state = self._joint_state
350:         if state is None:
351:             raise ValueError('joint state is incomplete')
352:         missing = set(REQUIRED_JOINT_NAMES) - set(state.name)
353:         if missing or len(state.position) != len(state.name):
354:             raise ValueError(f'joint state is incomplete: {sorted(missing)}')
355:         stamp_ns = state.header.stamp.sec * 1_000_000_000 + state.header.stamp.nanosec
356:         now_ns = self.get_clock().now().nanoseconds
357:         max_age_ns = int(float(self.get_parameter('joint_state_max_age_sec').value) * 1e9)
358:         if stamp_ns <= 0 or now_ns - stamp_ns > max_age_ns:
359:             raise TimeoutError('joint state is stale')
360:         return state

362:     def _visibility_constraint(self, candidate) -> VisibilityConstraint:
363:         frame = str(self.get_parameter('planning_frame').value)
364:         camera_frame = str(self.get_parameter('visibility_camera_frame').value)
365:         try:
366:             transform = self._visibility_tf.lookup_transform(
367:                 frame, camera_frame, Time(), timeout=Duration(seconds=1.0))
368:         except Exception as error:
369:             raise InfrastructureError(f'visibility camera TF unavailable: {error}') from error
370:         stamp = transform.header.stamp.sec*1_000_000_000 + transform.header.stamp.nanosec
371:         age = (self.get_clock().now().nanoseconds - stamp)/1e9
372:         maximum_age = float(self.get_parameter('visibility_camera_max_age_sec').value)
373:         if (not math.isfinite(maximum_age) or maximum_age <= 0.
374:                 or (stamp != 0 and not -0.05 <= age <= maximum_age)):
375:             raise InfrastructureError('visibility camera TF is stale or its age limit is invalid')
376:         camera = transform.transform.translation
377:         obj = candidate.target_object
378:         p, q, size = obj.obb_pose.position, obj.obb_pose.orientation, obj.obb_size
379:         cone = enclosing_visibility_cone(
380:             (camera.x, camera.y, camera.z), (p.x, p.y, p.z),
381:             (size.x, size.y, size.z), (q.x, q.y, q.z, q.w),
382:             padding_m=float(self.get_parameter('visibility_padding_m').value),
383:             sides=int(self.get_parameter('visibility_cone_sides').value))
384:         message = VisibilityConstraint()
385:         message.target_radius = cone.radius_m
386:         message.cone_sides = cone.sides
387:         message.target_pose.header.frame_id = frame
388:         message.target_pose.pose.position.x, message.target_pose.pose.position.y, message.target_pose.pose.position.z = cone.target
389:         (message.target_pose.pose.orientation.x, message.target_pose.pose.orientation.y,
390:          message.target_pose.pose.orientation.z, message.target_pose.pose.orientation.w) = cone.orientation
391:         message.sensor_pose.header.frame_id = frame
392:         message.sensor_pose.pose.position.x, message.sensor_pose.pose.position.y, message.sensor_pose.pose.position.z = cone.camera
393:         message.sensor_pose.pose.orientation.w = 1.
394:         message.weight = 1.
395:         self.get_logger().debug(f'Pregrasp visibility envelope radius={cone.radius_m:.4f}m camera={camera_frame}')
396:         return message

486:                         f'candidate={index} arm={arm} '
487:                         f'stage={stage.value}: {message}'
488:                     )
489: 
490:                 selection = self._selector.select(
491:                     candidates,
492:                     required_arm=getattr(
493:                         goal_handle.request, 'required_arm', ''
494:                     ),
495:                     cancel_requested=canceled,
496:                     feedback=feedback,
497:                 )
498:                 if selection is None:
499:                     self._set_failure(
500:                         result,
501:                         result.ERROR_NO_REACHABLE_GRASP,
502:                         'No candidate-arm pair passed IK, validity, and both plans',
503:                     )
504:                     self.get_logger().info('No reachable grasp selected')
505:                 else:
506:                     self.get_logger().info(
507:                         f'Selected reachable grasp: candidate={selection.candidate_index} arm={selection.arm}')
508:                     result.success = True
509:                     result.error_code = result.ERROR_NONE
510:                     result.message = 'Selected reachable grasp (plan-only)'
511:                     result.selected_candidate_index = selection.candidate_index
512:                     result.selected_arm = selection.arm
513:                     result.selected_candidate = candidate_messages[
514:                         selection.candidate_index
515:                     ]
516:                     result.pregrasp_joint_state = self._joint_message(
517:                         selection.pregrasp
518:                     )
519:                     result.grasp_joint_state = self._joint_message(
520:                         selection.grasp
521:                     )
522:                     terminal_state = 'succeed'
523:         except InterruptedError:
524:             self._set_failure(
525:                 result,
526:                 result.ERROR_CANCELED,
527:                 'Grasp selection canceled or timed out',
528:             )
529:             terminal_state = 'canceled'
530:         except InfrastructureError as error:
531:             message = str(error)
532:             code = self._infrastructure_error_code(result, message)
533:             self._set_failure(result, code, message)
534:         except (ValueError, TypeError) as error:
535:             self._set_failure(result, result.ERROR_INVALID_INPUT, str(error))
536:         except Exception as error:
537:             self._set_failure(
538:                 result,
539:                 result.ERROR_INTERNAL,
540:                 f'Unexpected error: {error}',
541:             )
542:         finally:
543:             if getattr(self, '_visibility_tf', None) is not None:
544:                 self._adapter.set_visibility_constraint(None)
545:             if scene_started:
546:                 try:
547:                     self._scene.restore()
548:                 except Exception as error:
549:                     self.get_logger().error(
550:                         f'Failed to restore planning scene: {error}'
551:                     )
552:                     self._set_failure(
553:                         result,
554:                         result.ERROR_PLANNING_SCENE,
555:                         f'Failed to restore planning scene: {error}',
556:                     )
557:                     terminal_state = 'abort'
558:             with self._goal_lock:
559:                 self._goal_active = False
560: 
561:         if terminal_state == 'succeed':
562:             goal_handle.succeed()
563:         elif terminal_state == 'canceled':
564:             goal_handle.canceled()
565:         else:
566:             goal_handle.abort()
567:         return result

569:     @staticmethod
570:     def _joint_message(solution) -> JointState:
571:         message = JointState()
572:         message.name = list(solution.names)
573:         message.position = list(solution.positions)
574:         return message
575: 
576:     @staticmethod

578:         result.success = False
579:         result.error_code = code
580:         result.message = message
581:         result.selected_candidate_index = -1
582:         result.selected_arm = ''
583:         result.selected_candidate = GraspCandidate()

585:         result.grasp_joint_state = JointState()
586: 
587:     @staticmethod
588:     def _infrastructure_error_code(result, message: str) -> int:
589:         planning_scene_terms = (
590:             'planning-scene',
591:             'planning scene',
592:             'target OBB',
593:             'collision permissions',
594:         )
595:         if any(term in message for term in planning_scene_terms):
596:             return result.ERROR_PLANNING_SCENE
597:         return result.ERROR_MOVEIT_UNAVAILABLE
