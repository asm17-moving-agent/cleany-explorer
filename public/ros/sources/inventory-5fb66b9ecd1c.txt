22: def wait(node: Node, future, timeout: float):
23:     deadline = time.monotonic() + timeout
24:     while rclpy.ok() and not future.done() and time.monotonic() < deadline:
25:         rclpy.spin_once(node, timeout_sec=0.05)
26:     if not future.done():
27:         raise TimeoutError('ROS response timed out; inspect the execution record before retrying')
28:     return future.result()

31: def require_execution_available(node: Node, prefix: str = '') -> None:
32:     """Read historical inhibition before spending time on scene inspection."""
33:     status = node.create_client(ConfirmManipulationRecovery, prefix + '/manipulation/confirm_recovery')
34:     try:
35:         if not status.wait_for_service(timeout_sec=2.):
36:             raise RuntimeError('Manipulation recovery status unavailable before inspection')
37:         # Both confirmation flags remain false: this request never clears a block.
38:         response = wait(node, status.call_async(ConfirmManipulationRecovery.Request()), 2.)
39:         if not response.success or response.inhibited:
40:             raise RuntimeError('Manipulation is inhibited; inspect and confirm recovery before scene inspection: '
