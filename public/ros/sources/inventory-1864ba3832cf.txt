64: validate_planning_mode() {
65:   case "${motion_backend}" in
66:     moveit) motion_runtime=native ;;
67:     curobo_ik_moveit) motion_runtime=container ;;
68:     *) echo "CLEANY_COLLECTION_MOTION_BACKEND must be moveit or curobo_ik_moveit." >&2; return 2 ;;
69:   esac
70:   case "${grasp_predictor}" in
71:     geometric|anygrasp) ;;
72:     *) echo "CLEANY_COLLECTION_GRASP_PREDICTOR must be geometric or anygrasp." >&2; return 2 ;;
73:   esac
74: }

76: require_matching_planning_mode() {
77:   # Runs started before mode selection used the licensed GPU pair.
78:   local active_mode='curobo_ik_moveit anygrasp'
79:   [[ ! -f "${planning_mode_file}" ]] || active_mode="$(<"${planning_mode_file}")"
80:   if [[ "${active_mode}" != "${motion_backend} ${grasp_predictor}" ]]; then
81:     echo "Running planning mode: ${active_mode}; requested: ${motion_backend} ${grasp_predictor}." >&2
82:     echo "Use make collection-stop, then restart with the requested planning options." >&2
83:     return 1
84:   fi
85: }

239: wait_for_graph() {
240:   check_managed_health || return 1
241:   local readiness_pid result
242:   local options=(--timeout "${start_timeout}")
243:   [[ "${gui}" != true ]] || options+=(--gui)
244:   # One participant discovers the existing requirements and waits for feedback
245:   # concurrently. The parent continues monitoring all managed processes.
246:   python3 "${repo_root}/ros2_ws/src/cleany_skill_executor/cleany_skill_executor/runtime_readiness.py" \
247:     "${options[@]}" &
248:   readiness_pid=$!
249:   while alive "${readiness_pid}"; do
250:     if ! check_managed_health; then
251:       kill -TERM "${readiness_pid}" 2>/dev/null || true
252:       wait "${readiness_pid}" 2>/dev/null || true
253:       return 1
254:     fi
255:     sleep .2
256:   done
257:   if wait "${readiness_pid}"; then result=0; else result=$?; fi
258:   (( result == 0 )) || return "${result}"
259:   check_managed_health || return 1
260: }

267: start_runtime() {
268:   validate_planning_mode || return
269:   if [[ "${gui}" != true && "${gui}" != false ]]; then
270:     echo "CLEANY_COLLECTION_GUI must be true or false." >&2
271:     return 2
272:   fi
273:   check_dds_transport || return
274:   mkdir -p "${state_dir}"
275:   if managed_processes_alive; then
276:     require_matching_planning_mode || return
277:     echo "Collection runtime is already running; repeating the full readiness check."
278:     source_ros
279:     wait_for_graph || return
280:     echo "COLLECTION RUNTIME READY"
281:     echo "Next: make collection-execute"
282:     return 0
283:   fi
284:   # The native CPU selector does not require a prepared cuRobo workspace.
285:   prepare_motion_runtime || return
286:   if fuser -s "${arm_port}"; then
287:     echo "A motor serial port is in use. Stop preview/control tools first." >&2
288:     fuser -v "${arm_port}" >&2 || true
289:     return 1
290:   fi
291:   if [[ -z "${GEMINI_API_KEY:-}" ]]; then
292:     echo "Load GEMINI_API_KEY into the environment before starting collection." >&2
293:     return 1
294:   fi
295:   "${repo_root}/tools/feetech-torque" check-off
296:   source_ros
297: 
298:   local timestamp run_dir log_dir
299:   timestamp="$(date +%Y%m%d-%H%M%S)"
300:   run_dir="${repo_root}/runtime/collection-operator/${timestamp}"
301:   log_dir="${run_dir}/logs"
302:   mkdir -p "${log_dir}" "${run_dir}/evidence"
303:   printf '%s\n' "${run_dir}" > "${run_dir_file}"
304:   printf '%s %s\n' "${motion_backend}" "${grasp_predictor}" > "${planning_mode_file}"
305:   : > "${pids_file}"
306:   rm -f "${pipeline_pid_file}" "${state_dir}/last_plan"
307: 
308:   export CLEANY_REFERENCE_OBSERVATION=true
309:   export CLEANY_WRIST_OBSERVATION=true
310:   export CLEANY_PERCEPTION_USE_SIM_TIME=false
311:   export CLEANY_PIPELINE_PROFILE=optimized
312:   if [[ "${gui}" == true ]]; then
313:     configure_gui_environment
314:   fi
315: 
316:   echo "Planning mode: ${grasp_predictor} candidates + ${motion_backend} (${motion_runtime})"
317:   start_vision "${log_dir}/vision.log" || return
318:   start_group left_wrist_camera "${log_dir}/left-wrist-camera.log" \
319:     ros2 run cleany_robot_interface wrist_rgb_camera --ros-args \
320:       -r __node:=left_wrist_rgb_camera \
321:       -p repository_root:="${repo_root}" \
322:       -p start_enabled:=false \
323:       -p config_file:="${repo_root}/configs/robot/left_wrist_rgb.json"
324:   start_group right_wrist_camera "${log_dir}/right-wrist-camera.log" \
325:     ros2 run cleany_robot_interface wrist_rgb_camera --ros-args \
326:       -r __node:=right_wrist_rgb_camera \
327:       -p repository_root:="${repo_root}" \
328:       -p start_enabled:=false \
329:       -p config_file:="${repo_root}/configs/robot/right_wrist_rgb.json"
330:   start_real_control "${log_dir}/real-control.log"
331:   start_group manipulation_evidence "${log_dir}/manipulation-evidence.log" \
332:     ros2 launch cleany_skill_executor manipulation_evidence.launch.py \
333:       contact_rule_file:="${repo_root}/configs/robot/gripper_contact_decision.yaml"
334:   start_group motor_telemetry "${log_dir}/motor-telemetry.jsonl" \
335:     ros2 run cleany_robot_interface motor_telemetry_recorder
336:   start_hardware_support "${log_dir}/hardware-support.log" "${run_dir}"
337:   if [[ "${gui}" == true ]]; then
338:     start_group bbox_view "${log_dir}/bbox-view.log" \
339:       ros2 run rqt_image_view rqt_image_view \
340:         --ros-args -r image:=/perception/debug_image_latched
341:   fi
342: 
343:   echo "Waiting for services, actions, 14-joint feedback, depth and OctoMap input..."
344:   if ! wait_for_graph; then
345:     echo "Startup preflight failed. Runtime is left running for inspection." >&2
346:     echo "Logs: ${log_dir}" >&2
347:     echo "Stop explicitly with: make collection-stop" >&2
348:     return 1
349:   fi
350:   echo "COLLECTION RUNTIME READY"
351:   echo "Torque: ON for the arm controllers"
352:   echo "MoveIt execution: OFF (plan-only)"
353:   if [[ "${gui}" == true ]]; then
354:     echo "RViz and bbox viewer: running"
355:   else
356:     echo "RViz and bbox viewer: disabled (headless)"
357:   fi
358:   echo "Logs/results: ${run_dir}"
359:   echo "Next: make collection-execute"
360: }

362: recover_control() {
363:   require_runtime || return
364:   local old_pid=''
365:   while read -r name pid; do
366:     if [[ "${name}" == real_control ]]; then
367:       old_pid="${pid}"
368:       break
369:     fi
370:   done < "${pids_file}"
371:   [[ -n "${old_pid}" ]] || {
372:     echo "Managed real_control PID is missing." >&2
373:     return 1
374:   }
375:   echo "Restarting latched real_control (PID ${old_pid}); motor torque is preserved"
376:   if alive "${old_pid}"; then
377:     kill -INT -- "-${old_pid}" 2>/dev/null || true
378:   fi
379:   local deadline=$((SECONDS + 20))
380:   while alive "${old_pid}" && (( SECONDS < deadline )); do
381:     sleep .25
382:   done
383:   if alive "${old_pid}"; then
384:     echo "real_control did not exit cleanly; torque-preserving recovery aborted." >&2
385:     return 1
386:   fi
387:   deadline=$((SECONDS + 10))
388:   while fuser -s "${arm_port}" && (( SECONDS < deadline )); do
389:     sleep .25
390:   done
391:   if fuser -s "${arm_port}"; then
392:     echo "Motor serial port remains in use; recovery aborted without changing torque." >&2
393:     fuser -v "${arm_port}" >&2 || true
394:     return 1
395:   fi
396: 
397:   local replacement="${pids_file}.recover.$$"
398:   awk '$1 != "real_control"' "${pids_file}" > "${replacement}" || return
399:   mv "${replacement}" "${pids_file}" || return
400:   rm -f "${state_dir}/last_plan"
401:   source_ros || return
402:   if [[ "${gui}" == true ]]; then
403:     configure_gui_environment
404:   fi
405:   local run_dir
406:   run_dir="$(current_run_dir)" || return
407:   start_real_control "${run_dir}/logs/real-control-recovery.log" || return
408:   echo "Waiting for torque-on hold adoption and controller readiness..."
409:   wait_for_graph || return
410:   echo "REAL CONTROL RECOVERED; TORQUE REMAINED ON"
411: }

462: run_foreground() {
463:   setsid "$@" &
464:   local child=$!
465:   printf '%s\n' "${child}" > "${pipeline_pid_file}"
466:   trap 'kill -INT -- -"${child}" 2>/dev/null || true' INT TERM
467:   local result=0
468:   wait "${child}" || result=$?
469:   trap - INT TERM
470:   rm -f "${pipeline_pid_file}"
471:   return "${result}"
472: }

474: plan_once() {
475:   require_runtime
476:   check_managed_health
477:   source_ros
478:   # Remove obsolete preview gates from older runtime versions.
479:   rm -f "${state_dir}/last_plan"
480:   local run_dir timestamp output
481:   run_dir="$(current_run_dir)"
482:   timestamp="$(date +%Y%m%d-%H%M%S)"
483:   output="${run_dir}/plan-${timestamp}.json"
484:   echo "PLAN ONLY — no trajectory command will be sent"
485:   if ! run_foreground ros2 run cleany_skill_executor pregrasp_pipeline \
486:       --query "${query}" --output "${output}" --ros-args \
487:       -p use_sim_time:=false -p require_sensor_scene:=true \
488:       -p sensor_scene_startup_timeout_sec:=60.0 -p pipeline_timeout_sec:=600.0; then
489:     echo "Plan failed. Result: ${output}" >&2
490:     [[ -f "${output}" ]] && python3 -m json.tool "${output}" || true
491:     return 1
492:   fi
493:   python3 - "${output}" <<'PY'
494: import json
495: import sys
496: from pathlib import Path
497: 
498: path = Path(sys.argv[1])
499: report = json.loads(path.read_text())
500: print(json.dumps(report, ensure_ascii=False, indent=2))
501: if not report.get("success") or report.get("mode") != "plan_only":
502:     raise SystemExit("Plan-only result is not successful")
503: if report.get("arm") not in ("left", "right"):
504:     raise SystemExit("Plan did not select an available arm")
505: PY
506:   echo "PLAN INSPECTION COMPLETE: ${output}"
507:   echo "Physical execution makes one fresh plan in its own request: make collection-execute"
508: }

510: execute_once() {
511:   require_runtime
512:   check_managed_health
513:   source_ros
514:   # Discard the obsolete preview gate. This explicit command performs one fresh
515:   # validation/selection request and executes that result in the same process.
516:   rm -f "${state_dir}/last_plan"
517:   local run_dir timestamp output
518:   run_dir="$(current_run_dir)"
519:   timestamp="$(date +%Y%m%d-%H%M%S)"
520:   output="${run_dir}/execute-${timestamp}.json"
521:   echo "PHYSICAL EXECUTION — select either arm, pick/lift/return/release at pickup (placement assumed)"
522:   if ! run_foreground ros2 run cleany_skill_executor collection_pipeline \
523:       --config "${CLEANY_MANIPULATION_COLLECTION_CONFIG}" \
524:       --query "${query}" --output "${output}" --execute --ros-args \
525:       --params-file "${CLEANY_HARDWARE_MONITOR_CONFIG}" \
526:       -p gripper_contact_rule_file:="${repo_root}/configs/robot/gripper_contact_decision.yaml" \
527:       -p use_sim_time:=false -p require_sensor_scene:=true \
528:       -p sensor_scene_startup_timeout_sec:=60.0 -p pipeline_timeout_sec:=600.0 \
529:       -p planning_scene_timeout_sec:=5.0 \
530:       -p hardware_path_step_rad:=0.01 \
531:       -p hardware_path_simplification_tolerance_rad:=0.002 \
532:       -p hardware_start_delay_sec:=0.8 \
533:       -p hardware_endpoint_settle_duration_sec:=0.3 \
534:       -p hardware_endpoint_settle_timeout_sec:=2.0 \
535:       -p hardware_endpoint_settle_velocity_rad_s:=0.02 \

548: start_action_server() {
549:   require_runtime || return
550:   check_managed_health || return
551:   if [[ -f "${pipeline_pid_file}" ]] && alive "$(<"${pipeline_pid_file}")"; then
552:     echo "The legacy collection pipeline is still running; wait for it to finish." >&2
553:     return 1
554:   fi
555:   local name pid
556:   while read -r name pid; do
557:     if [[ "${name}" == manipulation_action ]] && alive "${pid}"; then
558:       echo "Manipulation Action server is already running (PID ${pid})."
559:       return 0
560:     fi
561:   done < "${pids_file}"
562:   source_ros || return
563:   local services run_dir
564:   services="$(timeout 10 ros2 service list)" || return
565:   if ! require_lines "${services}" /perception/get_scene_snapshot; then
566:     echo "The perception snapshot service is missing. Rebuild and restart the perception runtime." >&2
567:     return 1
568:   fi
569:   run_dir="$(current_run_dir)" || return
570:   start_group manipulation_action "${run_dir}/logs/manipulation-action.log" \
571:     ros2 launch cleany_manipulation_bt manipulation_jetson.launch.py \
572:       repository_root:="${repo_root}" \
573:       shutdown_timeout_sec:="${execution_stop_timeout}" \
574:       collection_config:="${CLEANY_MANIPULATION_COLLECTION_CONFIG}" \
575:       hardware_monitor_config:="${CLEANY_HARDWARE_MONITOR_CONFIG}" || return
576:   local deadline=$((SECONDS + 30)) info=''
577:   while (( SECONDS < deadline )); do
578:     check_managed_health || return 1
579:     info="$(timeout 5 ros2 action info /manipulation/execute_skill 2>/dev/null)" || info=''
580:     if grep -Fqx 'Action servers: 1' <<<"${info}"; then
581:       break
582:     fi
583:     sleep .25
584:   done
585:   if ! grep -Fqx 'Action servers: 1' <<<"${info}"; then
586:     echo "The manipulation Action endpoint did not become uniquely available." >&2
587:     return 1
588:   fi
589:   echo "Manipulation Action server started; waiting for approved object Goals."
590:   echo "Log: ${run_dir}/logs/manipulation-action.log"
591: }

613: action_run_once() {
614:   local began=$SECONDS preparation_sec result=0
615:   action_execution_id="$(python3 -c 'import uuid; print(uuid.uuid4())')"
616:   export CLEANY_MANIPULATION_PROGRESS_FILE="${CLEANY_MANIPULATION_PROGRESS_FILE:-${XDG_STATE_HOME:-${HOME}/.local/state}/cleany/manipulation/request.json}"
617:   trap 'finish_action_request "$?"' EXIT
618:   action_request_progress --phase RUNTIME_PREPARING --message 'Preparing runtime, sensors and planning services'
619:   start_runtime
620:   action_request_progress --phase SERVER_PREPARING --message 'Preparing manipulation Action server'
621:   start_action_server
622:   source_ros
623:   preparation_sec=$((SECONDS - began))
624:   printf '{"event":"collection_command_prepared","wall_sec":%d,"clock_resolution_sec":1}\n' "${preparation_sec}"
625:   run_foreground ros2 run cleany_skill_executor manipulation_test_client \
626:     --execution-id "${action_execution_id}" \
627:     --namespace / --auto-select --mission-id operator-single-object --task-id pick-return-release \
628:     --inspection-query "${query}" --timeout 1200 \
629:     --collection-config "${CLEANY_MANIPULATION_COLLECTION_CONFIG}" \
630:     --ros-args --params-file "${repo_root}/ros2_ws/src/cleany_manipulation_bt/config/manipulation_jetson.yaml" || result=$?
631:   printf '{"event":"collection_command_timing","preparation_wall_sec":%d,"command_wall_sec":%d,"exit_code":%d,"clock_resolution_sec":1}\n' \
632:     "${preparation_sec}" "$((SECONDS - began))" "${result}"
633:   return "${result}"

644: stop_action_server() {
645:   require_runtime || return
646:   if [[ -f "${pipeline_pid_file}" ]] && alive "$(<"${pipeline_pid_file}")"; then
647:     echo "An execution client is still running; finish or cancel it before restarting the Action server." >&2
648:     return 1
649:   fi
650:   if [[ ! "${execution_stop_timeout}" =~ ^[1-9][0-9]*$ ]]; then
651:     echo "CLEANY_COLLECTION_EXECUTION_STOP_TIMEOUT_SEC must be a positive integer." >&2
652:     return 2
653:   fi
654:   local name pid old_pid=''
655:   while read -r name pid; do
656:     [[ "${name}" != manipulation_action ]] || old_pid="${pid}"
657:   done < "${pids_file}"
658:   if [[ -n "${old_pid}" ]] && alive "${old_pid}"; then
659:     echo "Stopping manipulation Action server (PID ${old_pid}); waiting for its execution owner"
660:     kill -INT -- "-${old_pid}" || return
661:     local deadline=$((SECONDS + execution_stop_timeout))
662:     while alive "${old_pid}" && (( SECONDS < deadline )); do
663:       sleep .25
664:     done
665:     if alive "${old_pid}"; then
666:       echo "Action server shutdown timed out; PID records retained. No replacement was started." >&2
667:       return 1
668:     fi
669:   fi
670:   local replacement="${pids_file}.action-restart.$$"
671:   awk '$1 != "manipulation_action"' "${pids_file}" > "${replacement}" || return
672:   mv "${replacement}" "${pids_file}" || return
673: }

688:     --control-log "/proc/${control_pid}/fd/1" "$@"

692:   # Drain the previous owner before restoring controllers. A stuck owner keeps
693:   # its PID and blocks replacement, just as for explicit action-restart.
694:   stop_action_server || return
695:   recover_control || return
696:   start_action_server

699: return_arms() {
700:   require_runtime || return
701:   check_managed_health || return
702:   if [[ -f "${pipeline_pid_file}" ]] && alive "$(<"${pipeline_pid_file}")"; then
703:     echo "An execution client is still running; finish or cancel it before arm return." >&2
704:     return 1
705:   fi
706:   source_ros || return
707:   local status=0 recovered=false
708:   arm_return_control_status "$@" || status=$?
709:   case "${status}" in
710:     0) start_action_server || return ;;
711:     20) recover_arm_return_control || return; recovered=true ;;
712:     *) return "${status}" ;;
713:   esac
714:   status=0
715:   run_foreground ros2 run cleany_manipulation_bt arm_return "$@" || status=$?
716:   # This code is returned only for a latched executor error BEFORE any motion.
717:   # Never repeat a failed opening, return trajectory, final stop or cancellation.
718:   if [[ "${status}" -eq 21 && "${recovered}" == false ]]; then
719:     status=0
720:     arm_return_control_status "$@" || status=$?
721:     case "${status}" in
722:       0) restart_action_server || return ;;
723:       20) recover_arm_return_control || return ;;
724:       *) return "${status}" ;;
725:     esac
726:     run_foreground ros2 run cleany_manipulation_bt arm_return "$@"
727:   else
728:     return "${status}"
729:   fi
730: }

740:     local name pid state
741:     while read -r name pid; do
742:       state=stopped
743:       alive "${pid}" && state=running

745:     done < "${pids_file}"

753:   docker ps --format '{{.Names}} {{.Status}}' | grep -F 'cleany-vision-' || true
754:   "${repo_root}/tools/motion-container" status --owner "${state_dir}" || true

763: signal_pipeline() {
764:   local stop_signal="${1:-INT}"
765:   [[ -f "${pipeline_pid_file}" ]] || return 0
766:   local pid
767:   pid="$(<"${pipeline_pid_file}")"
768:   if alive "${pid}"; then
769:     kill "-${stop_signal}" -- "-${pid}" 2>/dev/null || true
770:   fi
771: }

773: execution_owners_alive() {
774:   if [[ -f "${pipeline_pid_file}" ]] && alive "$(<"${pipeline_pid_file}")"; then
775:     return 0
776:   fi
777:   local name pid
778:   if [[ -f "${pids_file}" ]]; then
779:     while read -r name pid; do
780:       if [[ "${name}" == manipulation_action ]] && alive "${pid}"; then
781:         return 0
782:       fi
783:     done < "${pids_file}"
784:   fi
785:   return 1
786: }

788: wait_for_execution_shutdown() {
789:   local deadline=$((SECONDS + execution_stop_timeout))
790:   while execution_owners_alive; do
791:     (( SECONDS < deadline )) || return 1
792:     sleep 0.1
793:   done
794: }

796: stop_runtime() {
797:   if [[ ! "${execution_stop_timeout}" =~ ^[0-9]+$ ]] || (( execution_stop_timeout < 1 )); then
798:     echo "CLEANY_COLLECTION_EXECUTION_STOP_TIMEOUT_SEC must be a positive integer." >&2
799:     return 1
800:   fi
801:   mkdir -p "${state_dir}"
802:   signal_pipeline
803:   local target name pid
804:   if [[ -f "${pids_file}" ]]; then
805:     while read -r name pid; do
806:       if [[ "${name}" == manipulation_action ]] && alive "${pid}"; then
807:         echo "stopping ${name} (PID ${pid})"
808:         kill -INT -- "-${pid}" 2>/dev/null || true
809:       fi
810:     done < "${pids_file}"
811:   fi
812:   # Keep controllers, guard and observations alive while execution owners drain.
813:   if ! wait_for_execution_shutdown; then
814:     echo "Execution shutdown timed out; independent runtime shutdown follows. Physical stop is unconfirmed." >&2
815:   fi
816:   local stop_order=(bbox_view hardware_support manipulation_evidence left_wrist_camera right_wrist_camera real_control motor_telemetry vision)
817:   for target in "${stop_order[@]}"; do
818:     [[ -f "${pids_file}" ]] || break
819:     while read -r name pid; do
820:       if [[ "${name}" == "${target}" ]] && alive "${pid}"; then
821:         echo "stopping ${name} (PID ${pid})"
822:         kill -INT -- "-${pid}" 2>/dev/null || true
823:       fi
824:     done < "${pids_file}"
825:   done
826:   for _ in {1..20}; do
827:     managed_processes_alive || break
828:     sleep 0.5
829:   done
830:   if managed_processes_alive || execution_owners_alive; then
831:     signal_pipeline TERM
832:     if [[ -f "${pids_file}" ]]; then
833:       while read -r name pid; do
834:         if alive "${pid}"; then
835:           echo "terminating ${name} (PID ${pid})"
836:           kill -TERM -- "-${pid}" 2>/dev/null || true
837:         fi
838:       done < "${pids_file}"
839:     fi
840:     sleep 2
841:   fi
842:   "${repo_root}/tools/vision-container" hybrid-down || true
843:   "${repo_root}/tools/motion-container" stop --owner "${state_dir}" || true
844:   source_ros
845:   local daemon_stop_status=0
846:   timeout --signal=TERM --kill-after=1s "${daemon_stop_timeout}s" \
847:     ros2 daemon stop >/dev/null 2>&1 \
848:     || daemon_stop_status=$?
849:   if [[ "${daemon_stop_status}" -eq 124 || "${daemon_stop_status}" -eq 137 ]]; then
850:     echo "ROS 2 daemon stop timed out after ${daemon_stop_timeout}s; continuing shutdown." >&2
851:   elif [[ "${daemon_stop_status}" -ne 0 ]]; then
852:     echo "ROS 2 daemon stop failed (status ${daemon_stop_status}); continuing shutdown." >&2
853:   fi
854: 
855:   if fuser -s "${arm_port}"; then
856:     echo "Cannot release torque: a serial port is still in use." >&2
857:     fuser -v "${arm_port}" >&2 || true
858:     return 1
859:   fi
860:   "${repo_root}/tools/feetech-torque" off
861:   if managed_processes_alive || execution_owners_alive; then
862:     echo "Torque off verified, but execution/runtime processes remain; PID records retained." >&2
863:     return 1
864:   fi
865:   : > "${pids_file}"
866:   rm -f "${pipeline_pid_file}" "${state_dir}/last_plan"
867:   echo "COLLECTION RUNTIME STOPPED; TORQUE OFF VERIFIED FOR IDS 1-14"
868: }

872:   start|recover-control|recover-planning|action-start|action-restart|arm-return|action-run|execute|run)
873:     for profile_variable in CLEANY_MANIPULATION_COLLECTION_CONFIG CLEANY_FEETECH_CONTROL_CONFIG CLEANY_HARDWARE_MONITOR_CONFIG; do
874:       if [[ -z "${!profile_variable:-}" || ! -f "${!profile_variable}" ]]; then
875:         echo "Set ${profile_variable} to an explicit local hardware profile." >&2
876:         exit 2
877:       fi
878:     done
879:     ;;
880: esac
881: operator_directory="${CLEANY_WEB_CONTROL_DIRECTORY:-${XDG_STATE_HOME:-${HOME}/.local/state}/cleany/dev-monitor/operator}"
882: if [[ -f "${operator_directory}/enabled" && "${CLEANY_CONTROL_RUNTIME_LOCKED:-0}" != 1 ]]; then
883:   case "${command}" in
884:     start|recover-control|recover-planning|action-start|action-restart|arm-return|action-run|execute|run|plan)
885:       mkdir -p "${operator_directory}"
886:       # --close keeps background ROS processes from inheriting the command lock.
887:       exec flock -n -E 75 --close "${operator_directory}/operation.lock" \
888:         env CLEANY_CONTROL_RUNTIME_LOCKED=1 "$0" "$@"
889:       ;;
890:   esac
891: fi
892: case "${command}" in
893:   start) start_runtime ;;
894:   recover-control) recover_control ;;
895:   recover-planning) recover_planning ;;
896:   action-start) start_action_server ;;
897:   action-recover) shift; confirm_action_recovery "$@" ;;
898:   action-restart) restart_action_server ;;
899:   arm-return) shift; return_arms "$@" ;;
900:   action-run) action_run_once ;;
901:   plan) plan_once ;;
902:   execute) execute_once ;;
903:   run) run_once ;;
904:   status) status_runtime ;;
905:   logs) follow_logs ;;
906:   stop) stop_runtime ;;
907:   help|-h|--help|'') usage; [[ -n "${command}" ]] ;;
908:   *) usage; exit 2 ;;
909: esac
