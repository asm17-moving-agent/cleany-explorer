22: class BaseDriver(Node):
23:     """Translate chassis commands into the MCU wheel permission protocol.
24: 
25:     Subscriptions:
26:         cmd_vel (geometry_msgs/msg/Twist): Requested body-frame planar velocity.
27:         base/wheel_state (cleany_base_interfaces/msg/WheelState): MCU feedback
28:             including boot identity, command acknowledgement and encoder counts.
29: 
30:     Publishers:
31:         base/wheel_command (cleany_base_interfaces/msg/WheelCommand): Enable,
32:             stop or expiring wheel-velocity commands, in FL/FR/RL/RR order.
33:         joint_states (sensor_msgs/msg/JointState): Unwrapped wheel angles in
34:             radians and measured angular velocities.
35:         diagnostics (diagnostic_msgs/msg/DiagnosticArray): Feedback health.
36: 
37:     Services:
38:         base/enable (std_srvs/srv/SetBool): Request enable or immediate stop.
39:             A successful enable response means sent, not yet acknowledged.
40: 
41:     QoS:
42:         Wheel transport uses best-effort, volatile, depth 1 in both directions.
43: 
44:     Notes:
45:         Starts disabled; geometry and limits must be supplied. Commands remain
46:         blocked until MCU acknowledgement. Steady-clock checks stop on stale
47:         feedback, command expiry, reboot or incompatible feedback.
48:     """
49:     def __init__(self, parameter_overrides=None) -> None:
50:         super().__init__('cleany_base_driver', parameter_overrides=parameter_overrides or [])
51:         self.declare_parameter('mock', False)
52:         for key, default in (('geometry.wheel_radius_m', 0.), ('geometry.wheelbase_m', 0.),
53:                              ('geometry.wheel_separation_m', 0.)):
54:             self.declare_parameter(key, default)
55:         for key in ('linear_x_mps', 'linear_y_mps', 'angular_z_rad_s', 'wheel_rad_s', 'command_timeout_s'):
56:             self.declare_parameter('limits.' + key, 0.)
57:         self.declare_parameter('counts_per_revolution', 3172)
58:         self.geometry = Geometry(*(float(self.get_parameter('geometry.' + p).value)
59:                                    for p in ('wheel_radius_m', 'wheelbase_m', 'wheel_separation_m')))
60:         self.limits = Limits(*(float(self.get_parameter('limits.' + p).value)
61:                                for p in ('linear_x_mps', 'linear_y_mps', 'angular_z_rad_s',
62:                                          'wheel_rad_s', 'command_timeout_s')))
63:         self.encoder = EncoderAdapter(int(self.get_parameter('counts_per_revolution').value))
64:         self.command_qos = QoSProfile(depth=1, reliability=ReliabilityPolicy.BEST_EFFORT,
65:                                        durability=DurabilityPolicy.VOLATILE)
66:         self.publisher = self.create_publisher(WheelCommand, 'base/wheel_command', self.command_qos)
67:         self.joints = self.create_publisher(JointState, 'joint_states', 10)
68:         self.diags = self.create_publisher(DiagnosticArray, 'diagnostics', 10)
69:         self.create_subscription(WheelState, 'base/wheel_state', self.on_state, self.command_qos)
70:         self.create_subscription(Twist, 'cmd_vel', self.on_twist, 10)
71:         self.create_service(SetBool, 'base/enable', self.on_enable)
72:         self.enabled = False
73:         self.latest = self.last_state_receipt = None
74:         self.last_cmd_receipt = self.last_state_sequence = self.last_mcu_timestamp = None
75:         self.latest_twist = None
76:         self.boot_id = None
77:         self.command_sequence = None
78:         self.pending_enable = None
79:         self.diagnostic_error = ''
80:         self._stop_sent = False
81:         self.create_timer(.02, self.tick, clock=Clock(clock_type=ClockType.STEADY_TIME))
82:         self.create_timer(.1, self.publish_diagnostics, clock=Clock(clock_type=ClockType.STEADY_TIME))
83: 
84:     def _deadline(self) -> int:
85:         return self._mcu_now() + min(200000, int(self.limits.timeout_s * 1e6))
86: 
87:     def _mcu_now(self) -> int:
88:         return self.latest.timestamp_us + max(0, int((time.monotonic() - self.last_state_receipt) * 1e6))
89: 
90:     def _send(self, mode: int, values=(0., 0., 0., 0.), sequence=None, deadline=0) -> WheelCommand:
91:         msg = WheelCommand()
92:         msg.protocol_version = 2
93:         if sequence is None:
94:             self.command_sequence = ((self.command_sequence or 0) + 1) & 0xffffffff
95:             sequence = self.command_sequence
96:         msg.sequence, msg.mode = sequence, mode
97:         msg.velocity_rad_s = list(values)
98:         msg.valid_until_us = deadline
99:         self.publisher.publish(msg)
100:         return msg
101: 
102:     def _disable(self, reason: str, send_stop=True) -> None:
103:         self.enabled = False
104:         self.pending_enable = None
105:         self.last_cmd_receipt = None
106:         self.latest_twist = None
107:         self.diagnostic_error = reason
108:         if send_stop and not self._stop_sent:
109:             self._send(WheelCommand.STOP)
110:             self._stop_sent = True
111: 
112:     def on_enable(self, request, response):
113:         if not request.data:
114:             self._disable('stopped by request')
115:             response.success, response.message = True, 'STOP sent'
116:             return response
117:         if self.enabled:
118:             response.success, response.message = True, 'already enabled'
119:             return response
120:         if self.latest is None or self.last_state_receipt is None or time.monotonic() - self.last_state_receipt > .25:
121:             response.success, response.message = False, 'no fresh wheel feedback'
122:             return response
123:         if self.command_sequence is None:
124:             self.command_sequence = self.latest.last_command_sequence
125:         elif newer(self.latest.last_command_sequence, self.command_sequence):
126:             self.command_sequence = self.latest.last_command_sequence
127:         self.command_sequence = (self.command_sequence + 1) & 0xffffffff
128:         deadline = self._deadline()
129:         self.pending_enable = (self.command_sequence, deadline)
130:         self.enabled = True
131:         self._stop_sent = False
132:         self.last_cmd_receipt = None
133:         self.latest_twist = None
134:         self._send(WheelCommand.ENABLE, sequence=self.command_sequence, deadline=deadline)
135:         response.success, response.message = True, 'ENABLE sent; awaiting MCU acknowledgement'
136:         return response

138:     def on_twist(self, msg: Twist) -> None:
139:         axes = (msg.linear.x, msg.linear.y, msg.linear.z, msg.angular.x, msg.angular.y, msg.angular.z)
140:         if not finite_twist_axes(*axes):
141:             if self.enabled:
142:                 self._disable('non-finite cmd_vel')
143:             return
144:         if not self.enabled:
145:             return
146:         self.last_cmd_receipt = time.monotonic()
147:         self.latest_twist = (msg.linear.x, msg.linear.y, msg.angular.z)
148:         if any(v != 0 for v in (msg.linear.z, msg.angular.x, msg.angular.y)):
149:             self.get_logger().warning('unsupported cmd_vel axes ignored', throttle_duration_sec=2)
150: 
151:     def on_state(self, msg: WheelState) -> None:
152:         """Validate feedback before refreshing health or advancing enable state.
153: 
154:         Duplicate sequences and non-increasing MCU timestamps do not refresh the
155:         watchdog. A boot change disarms motion and rebases encoder deltas; feedback
156:         must acknowledge the pending enable sequence before velocity can be sent.
157:         """
158:         now = time.monotonic()
159:         if msg.protocol_version != 2 or msg.boot_id == 0 or msg.counts_per_revolution != self.encoder.scale or \
160:                 msg.max_velocity_rad_s != 10.0 or msg.watchdog_ms != 250:
161:             self._disable('WheelState contract mismatch')
162:             return
163:         if self.boot_id == msg.boot_id and self.last_state_sequence is not None and \
164:                 not newer(msg.sequence, self.last_state_sequence):
165:             return
166:         if self.boot_id == msg.boot_id and self.last_mcu_timestamp is not None and \
167:                 msg.timestamp_us <= self.last_mcu_timestamp:
168:             return
169:         reboot = self.boot_id is not None and self.boot_id != msg.boot_id
170:         if reboot:
171:             self.command_sequence = msg.last_command_sequence
172:             self._stop_sent = False
173:             self._disable('MCU reboot detected')
174:             self.last_state_sequence = self.last_mcu_timestamp = None
175:             self.encoder.rebase()
176:         if not all(isfinite(float(v)) for v in (*msg.velocity_rad_s, *msg.target_rad_s, *msg.commanded_rad_s)):
177:             self._disable('non-finite wheel feedback')
178:             return
179:         self.boot_id = msg.boot_id
180:         self.last_state_sequence = msg.sequence
181:         self.last_mcu_timestamp = msg.timestamp_us
182:         self.latest, self.last_state_receipt = msg, now
183:         if self.command_sequence is None or newer(msg.last_command_sequence, self.command_sequence):
184:             self.command_sequence = msg.last_command_sequence
185:         if self.pending_enable is not None:
186:             seq, _ = self.pending_enable
187:             if msg.last_command_sequence == seq and msg.enabled:
188:                 self.pending_enable = None
189:                 self.diagnostic_error = ''
190:         if self.enabled and not msg.enabled and self.pending_enable is None:
191:             self._disable('MCU unexpectedly disabled')
192:         elif not self.enabled and msg.enabled:
193:             self._disable('MCU enabled without host permission')
194:         if msg.fault_bits:
195:             if self.pending_enable is None:
196:                 self._disable(f'MCU fault bits: {msg.fault_bits:#x}')
197:         positions = self.encoder.update(msg.boot_id, tuple(msg.encoder_counts))
198:         out = JointState()
199:         out.header.stamp = self.get_clock().now().to_msg()
200:         out.name = ['front_left_wheel_joint', 'front_right_wheel_joint',
201:                     'rear_left_wheel_joint', 'rear_right_wheel_joint']
202:         out.position, out.velocity = list(positions), [float(v) for v in msg.velocity_rad_s]
203:         self.joints.publish(out)
204: 
205:     def tick(self) -> None:
206:         now = time.monotonic()
207:         if self.last_state_receipt is None or now - self.last_state_receipt > .25:
208:             if self.enabled:
209:                 self._disable('stale wheel feedback')
210:             if self.latest is not None:
211:                 self.latest = self.last_state_receipt = None
212:                 self.encoder.rebase()
213:             return
214:         if not self.enabled:
215:             return
216:         if self.last_cmd_receipt is not None and now - self.last_cmd_receipt > self.limits.timeout_s:
217:             self._disable('cmd_vel timeout')
218:             return
219:         if self.pending_enable is not None:
220:             seq, deadline = self.pending_enable
221:             if self._mcu_now() >= deadline:
222:                 self._disable('ENABLE acknowledgement timeout')
223:                 return
224:             self._send(WheelCommand.ENABLE, sequence=seq, deadline=deadline)
225:             return
226:         if not self.latest.enabled:
227:             return
228:         twist = self.latest_twist
229:         wheels = (0.,) * 4 if twist is None else wheel_speeds(*twist, self.geometry, self.limits)
230:         self._send(WheelCommand.VELOCITY, wheels, deadline=self._deadline())
231: 
232:     def publish_diagnostics(self) -> None:
233:         status = DiagnosticStatus()
234:         age = float('inf') if self.last_state_receipt is None else time.monotonic() - self.last_state_receipt
235:         status.name, status.hardware_id = 'base_driver', 'micro_ros_base'
236:         status.level = DiagnosticStatus.ERROR if age > .25 or self.diagnostic_error else DiagnosticStatus.OK
237:         status.message = self.diagnostic_error or ('stale wheel feedback' if age > .25 else 'feedback current')
238:         status.values = [KeyValue(key='enabled', value=str(self.enabled)), KeyValue(key='feedback_age_s', value=str(age))]
239:         arr = DiagnosticArray()
240:         arr.header.stamp, arr.status = self.get_clock().now().to_msg(), [status]
241:         self.diags.publish(arr)

244: def main(args=None):
245:     rclpy.init(args=args, signal_handler_options=SignalHandlerOptions.NO)
246:     node = BaseDriver()
247:     try:
248:         rclpy.spin(node)
249:     except (KeyboardInterrupt, ExternalShutdownException):
250:         pass
251:     finally:
252:         if rclpy.ok():
253:             node._disable('driver shutdown')
254:         node.destroy_node()
255:         if rclpy.ok():
256:             rclpy.shutdown()
