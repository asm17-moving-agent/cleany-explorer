132:     def _control_state(self, _request, response):
133:         """Report current executor ownership without commanding or clearing faults."""
134:         with self.core.lock:
135:             record = self.core.record
136:             active = record is not None and record.result is None and record.record_state == RecordState.ACTIVE
137:             operations = self.operations
138:             real = getattr(operations, 'real', None)
139:             guard = getattr(real, 'guard', None)
140:             guard_time = getattr(real, 'guard_time', None)
141:             max_age = real.value('hardware_guard_max_age_sec') if real is not None else 1.
142:             fresh = guard_time is not None and time.monotonic() - guard_time < max_age
143:             response.message = json.dumps(dict(schema=1,
144:                 active_execution_id=record.goal.execution_id if active else '',
145:                 active_stage=record.stage.value if active else '',
146:                 active_substage=record.substage if active else '',
147:                 arm_return_active=self.core.arm_return_active,
148:                 recovery_active=self.core._recovery_in_progress,

168:     def _confirm_recovery(self, request, response):
169:         if not request.robot_stopped_confirmed and not request.gripper_empty_confirmed:
170:             response.success, response.message = True, 'Recovery status; no state changed'
171:         else:
172:             verify = getattr(self.operations, 'confirm_recovery_stationary', None)
173:             if verify is None:
174:                 response.success, response.message = False, 'Physical stop verification unavailable'
175:             else:
176:                 response.success, response.message = self.core.confirm_recovery(
177:                     request.execution_id, request.expected_revision,
178:                     stopped=request.robot_stopped_confirmed,

181:                 if response.success:

190:     def _tick(self) -> None:
191:         with self.core.lock:
192:             if self._closing:
193:                 return
194:             if self.core.record is not None:
195:                 super()._tick()
196:                 self._update_operation_progress()
197:             if self.core.tree is not None:
198:                 execution = self.core.record.goal.execution_id
199:                 if execution != self._monitor_failed_execution:
200:                     try:
201:                         self.monitor.update(self.core.tree.snapshot(), new_execution=execution != self._monitor_execution)
202:                     except Exception as error:
203:                         self._monitor_failed_execution = execution
204:                         self.get_logger().warning(f'BT monitor unavailable for {execution}: {error}')
205:                     else:
206:                         self._monitor_execution = execution
207:                 for event in self.core.transitions:
208:                     self._bt_events.publish(String(data=json.dumps(dict(execution_id=execution, **event))))
209:                 self.core.transitions.clear()
210:             for _ in range(10):
211:                 try:
212:                     frames = self.socket.recv_multipart(flags=zmq.NOBLOCK)
213:                 except zmq.Again:

215:                 self.socket.send_multipart(self.monitor.reply(frames))

228:         core = getattr(self, 'core', None)
229:         if core is not None:
230:             with core.lock:
231:                 self._closing = True
232:         arm_return = getattr(self, 'arm_return', None)
233:         if arm_return is not None:
234:             arm_return.destroy()
235:         timer, action, core = (getattr(self, name, None) for name in ('_timer', '_action', 'core'))
236:         if timer is not None:
237:             timer.cancel()
238:         progress_timer = getattr(self, '_progress_timer', None)
239:         if progress_timer is not None:
240:             progress_timer.cancel()
241:         if action is not None:
242:             action.destroy()
243:         # Shutdown only halts work; the ACTIVE record remains for recovery.
244:         if core is not None and core.tree is not None and core.record.result is None:
245:             core.tree.halt()
246:         if self.backend is not None:
247:             self.backend.close()
248:         if self.operations is not None:
249:             self.operations.destroy_node()
250:         for name, method in (('socket', 'close'), ('zmq_context', 'term'), ('store', 'close')):
251:             resource = getattr(self, name, None)

257: def drain_shutdown(node: JetsonManipulationNode, executor: MultiThreadedExecutor) -> bool:
258:     """Keep DDS alive for IMMEDIATE cancellation and independent stop assessment."""
259:     with node.core.lock:
260:         node.core.inhibited = True
261:     arm_return = getattr(node, 'arm_return', None)
262:     if arm_return is not None:
263:         arm_return.request_shutdown()
264:         if node.core.arm_return_active:
265:             if not arm_return.drain(executor, node.core.stop_timeout_sec + 2.):
266:                 return False
267:     with node.core.lock:
268:         record = node.core.record
269:         if record is None or record.result is not None:
270:             return True
271:         execution_id = record.goal.execution_id
272:     request = CancelManipulation.Request(execution_id=execution_id, mode='IMMEDIATE')
273:     deadline = time.monotonic() + node.core.stop_timeout_sec + 2.0
274:     task = receipt = None
275:     # The application future completing precedes rclpy dispatching its Result
276:     # responses. A standard GetResult round trip keeps those callbacks alive.
277:     result_client = node.create_client(ExecuteManipulationSkill.Impl.GetResultService,
278:                                        'manipulation/execute_skill/_action/get_result')
279:     try:
280:         while node.context.ok() and time.monotonic() < deadline:
281:             with node.core.lock:
282:                 record = node.core.record
283:                 if (record.result is not None and node._handle is None and not node._futures
284:                         and receipt is not None and receipt.done() and receipt.exception() is None):
285:                     return record.result.stop_confirmed
286:                 if node._handle is not None:
287:                     if receipt is None:
288:                         receipt = result_client.call_async(ExecuteManipulationSkill.Impl.GetResultService.Request(
289:                             goal_id=node._handle.goal_id))
290:                     if task is None or (task.done() and (
291:                             task.exception() is not None or not task.result().accepted)):
292:                         task = executor.create_task(node._request_cancel, request, CancelManipulation.Response())
293:             executor.spin_once(timeout_sec=0.05)
294:         node.backend.abort()
295:         node.get_logger().error('Shutdown stop unconfirmed; inspect the execution journal and physical state')
296:         return False
297:     finally:
298:         node.destroy_client(result_client)

301: def main(args=None) -> None:
302:     # ROS's default handlers shut the context down before controller cancellation
303:     # can receive its acknowledgement or fresh feedback. Retain DDS until drained.
304:     rclpy.init(args=args, signal_handler_options=SignalHandlerOptions.NO)
305:     shutdown = threading.Event()
306:     previous_handlers = {sig: signal.signal(sig, lambda *_: shutdown.set())
307:                          for sig in (signal.SIGINT, signal.SIGTERM)}
308:     node = None
309:     executor = MultiThreadedExecutor(num_threads=4)
310:     try:
311:         node = JetsonManipulationNode()
312:         executor.add_node(node)
313:         if node.operations is not None:
314:             executor.add_node(node.operations)
315:         while rclpy.ok() and not shutdown.is_set():
316:             executor.spin_once(timeout_sec=0.1)
317:     except (KeyboardInterrupt, ExternalShutdownException):
318:         pass
319:     finally:
320:         try:
321:             try:
322:                 if node is not None and rclpy.ok():
323:                     drain_shutdown(node, executor)
324:             finally:
325:                 # launch may repeat signals while resources are being released.
326:                 for sig in previous_handlers:
327:                     signal.signal(sig, signal.SIG_IGN)
328:                 executor.shutdown()
329:                 if node is not None:
330:                     node.destroy_node()
331:                 if rclpy.ok():
332:                     rclpy.shutdown()
333:         finally:
334:             for sig, handler in previous_handlers.items():
335:                 signal.signal(sig, handler)
