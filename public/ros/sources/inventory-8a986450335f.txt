432:         tolerance = float(self.get_parameter('maximum_target_translation_m').value)
433:         if (observation.header.frame_id != self.config['frame']

537:         lift_collisions = self.config.get('lift_collision_check_enabled', True)

541:         with self.record_pickup_departure(arm):

685:     def _assess_stationary(self, context, *, cancel):
686:         started = time.monotonic()
687:         barrier = self.get_clock().now().nanoseconds
688:         window = StationaryWindow(barrier_ns=barrier, started_at=started,
689:             max_age_sec=float(self.value('bt_stop_feedback_max_age_sec')),
690:             maximum_velocity=float(self.value('bt_stationary_velocity_rad_s')),
691:             required_samples=int(self.value('bt_stationary_samples')),
692:             minimum_duration_sec=float(self.value('bt_stationary_duration_sec')))
693:         canceled = set()
694:         while time.monotonic() - started < self.value('bt_stop_timeout_sec'):
695:             if cancel:
696:                 for handle, result in list(self.handles):
697:                     if not result.done() and id(handle) not in canceled:
698:                         try:
699:                             handle.cancel_goal_async()
700:                         except Exception as error:
701:                             self.get_logger().error(f'Could not request Action cancellation: {error}')
702:                             self.real.fail_stop()
703:                             return False
704:                         canceled.add(id(handle))
705:             with self._feedback_lock:
706:                 confirmed = window.observe(self._feedback, now=time.monotonic(),
707:                     clock_ns=self.get_clock().now().nanoseconds, terminal=self.quiescent())
708:             if confirmed:
709:                 context.stop_confirmed = True
710:                 return True
711:             time.sleep(0.01)
712:         if cancel:
713:             self.real.fail_stop()
714:         return False

716:     def stop(self, context):
717:         context.abort.set()

719:         if confirmed and self.carrying is None:
720:             self.scene.restore()
721:             self.release_grasp_scene_freeze()
722:             if not context.gripper_engaged or context.release_confirmed:
723:                 self._clear_references(context)
724:         return Observation(confirmed, Error.NONE if confirmed else Error.STOP_UNCONFIRMED,
725:             'Fresh joints and terminal Actions confirmed stop' if confirmed else 'Physical stop unconfirmed',
726:             stop_confirmed=confirmed)
