23:         self._requested_arms: set[str] = set()

26:     def _set(self, arm: str, enabled: bool, *, cleanup: bool = False) -> None:
27:         client = self.clients[arm]
28:         if not client.wait_for_service(timeout_sec=self.timeout_seconds):
29:             raise RuntimeError(f'{arm} wrist camera control service unavailable')
30:         future = client.call_async(SetBool.Request(data=enabled))
31:         self._acknowledge(arm, future, cleanup=cleanup)
32: 
33:     def _acknowledge(self, arm: str, future: Any, *, cleanup: bool) -> None:
34:         if cleanup:
35:             # Closing capture must remain possible after the motion guard has aborted.
36:             deadline = time.monotonic() + self.timeout_seconds
37:             while not future.done() and time.monotonic() < deadline:
38:                 time.sleep(.01)
39:             if not future.done():
40:                 raise RuntimeError(f'Timed out disabling {arm} wrist camera')
41:             response = future.result()
42:         else:
43:             response = self.wait(future, self.timeout_seconds)
44:         if response is None or not response.success:
45:             message = response.message if response is not None else 'No response'
46:             raise RuntimeError(f'{arm} wrist camera control failed: {message}')

48:     def activate(self, arm: str) -> None:
49:         if arm not in self.clients:
50:             raise ValueError('Unknown wrist camera arm')
51:         if self.active_arm == arm and arm not in self._disable_pending:
52:             return
53:         other = 'right' if arm == 'left' else 'left'
54:         # Require the opposite capture to be closed before opening the selected one.
55:         self._requested_arms.add(other)
56:         self._set(other, False)
57:         self._requested_arms.discard(other)
58:         self._disable_pending.pop(other, None)
59:         if self.active_arm == other:
60:             self.active_arm = None
61:         self._requested_arms.add(arm)
62:         self._disable_pending.pop(arm, None)
63:         self._set(arm, True)
64:         self.active_arm = arm

66:     def close(self) -> None:
67:         errors = []
68:         for arm in sorted(self._requested_arms):
69:             try:
70:                 pending = self._disable_pending.pop(arm, None)
71:                 if pending is None:
72:                     self._set(arm, False, cleanup=True)
73:                 else:
74:                     self._acknowledge(arm, pending, cleanup=True)
75:             except Exception as error:

78:                 self._requested_arms.discard(arm)
79:                 if self.active_arm == arm:
80:                     self.active_arm = None
81:         if errors:
82:             raise RuntimeError('; '.join(errors))
