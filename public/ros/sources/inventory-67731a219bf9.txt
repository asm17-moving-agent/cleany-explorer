246:     def _execution_goal(self, arm, joint_state, label):
247:         goal = super()._execution_goal(arm, joint_state, label)
248:         if f'{arm}_gripper_joint' in joint_state.name:
249:             expected = {name for name in REQUIRED_JOINT_NAMES if name.startswith(f'{arm}_')}
250:             if arm not in ('left', 'right') or set(joint_state.name) != expected or self._held_object is not None:
251:                 raise ValueError('Combined pregrasp requires only the selected empty arm and gripper')
252:             # One collision-checked six-joint path, dispatched by MoveIt to
253:             # disjoint arm/gripper controllers. No concurrent blind open command.

337:     def _plan_linear_motion(self, arm: str, target: Pose, label: str, **kwargs) -> RobotTrajectory:
338:         if not getattr(self, '_carry_wrist_reference', {}) or self._held_object is None:
339:             return super()._plan_linear_motion(arm, target, label, **kwargs)
340:         self._wait_arm_stationary(arm)
341:         start = self._tcp_pose(arm)
342:         position = np.array(self._pose_position(target))
343:         failures = []
344:         for degrees in self.get_parameter('sorting_carry_wrist_tolerances_deg').value:
345:             tolerance = math.radians(degrees)
346:             if tolerance+1e-9 < self._carry_wrist_tolerance:
347:                 continue
348:             self._transport_adapter.set_current_state(self._feedback_state())
349:             solution = self._transport_adapter.solve_held_region_ik(
350:                 arm, position-1e-4, position+1e-4, np.zeros(3),
351:                 attempts=int(self.get_parameter('sorting_release_ik_attempts').value),
352:                 iterations=int(self.get_parameter('sorting_release_ik_iterations').value),
353:                 joint_bounds={name: (value-tolerance, value+tolerance)
354:                               for name, value in self._carry_wrist_reference.items()})
355:             if solution is None:
356:                 reason = 'no bounded endpoint IK'
357:             else:
358:                 joints = JointState(name=list(solution.names), position=list(solution.positions))
359:                 actual = self._tcp_pose(arm, joints)
360:                 try:
361:                     # A fixed wrist can change the TCP world orientation as
362:                     # shoulder/elbow move. Do not reuse the empty-arm pregrasp
363:                     # orientation as a mandatory carry endpoint. Bound the new
364:                     # rotation, then validate the whole FK corridor below.
365:                     orientation_reference = deepcopy(start)
366:                     orientation_reference.position = deepcopy(target.position)
367:                     validate_pose_endpoint(
368:                         self._cartesian_pose(actual), self._cartesian_pose(orientation_reference),
369:                         float(self.get_parameter('lin_position_tolerance_m').value),
370:                         math.radians(float(self.get_parameter('sorting_carry_cartesian_rotation_limit_deg').value)))
371:                 except ValueError as error:
372:                     reason = str(error)
373:                 else:
374:                     self._carry_wrist_tolerance = tolerance
375:                     options = dict(kwargs, joint_target=joints)
376:                     try:
377:                         trajectory = super()._plan_linear_motion(arm, actual, label, **options)
378:                     except CartesianPlanningError as error:
379:                         reason = str(error)
380:                     else:
381:                         self.get_logger().debug(
382:                             f'Carry Cartesian plan verified: {label} wrist=+/-{degrees:.1f}deg')

385:             self.get_logger().info(f'Carry Cartesian replan: {label} {failures[-1]}')
386:         raise CartesianPlanningError(f'{label}: no bounded carry path; ' + '; '.join(failures))

420:     def run(self):
421:         self._stage('starting')
422:         self._wait_for_pipeline()
423:         verify_placement = getattr(self, '_placement_verification_enabled', True)
424:         if verify_placement and not self._verification.wait_for_service(timeout_sec=10.0):
425:             raise RuntimeError('Placement verification port is unavailable')
426:         self._register_bins()

428:             if not self._wrist_client.wait_for_service(timeout_sec=10):
429:                 raise RuntimeError('Wrist observation service unavailable')
430:             self._switch_camera('head')
431:         self._home = {arm: self._arm_joint_state(arm)
432:                       for arm in ('left', 'right')}
433:         self._stage('search')
434:         maximum = int(self.get_parameter('sorting_maximum_objects').value)
435:         empty_confirmations = 0
436:         minimum_total_placements = 0
437:         for _ in range(maximum):

604:     def _ensure_fresh_head_before_wrist(self, selected, attempt):
605:         maximum = float(self.get_parameter('sorting_head_reference_refresh_age_sec').value)
606:         if not math.isfinite(maximum) or not 0 <= maximum <= 30.0:
607:             raise ValueError('Head reference refresh age must be in [0, 30] seconds')
608:         if maximum == 0.0:
609:             return selected, attempt
610:         def age(selection):
611:             stamp = selection.selected_candidate.header.stamp
612:             captured = stamp.sec*10**9 + stamp.nanosec
613:             now = self.get_clock().now().nanoseconds
614:             if captured <= 0 or captured > now:
615:                 raise RuntimeError('Invalid head reference timestamp before wrist handoff')
616:             return (now-captured)/1e9
617:         if age(selected) <= maximum:
618:             return selected, attempt
619:         self.get_logger().info('Refreshing aged head reference before wrist handoff')
620:         self._switch_camera('head')
621:         refreshed, fresh_attempt = self._refresh_selected_grasp(selected, attempt)
622:         if age(refreshed) > maximum:
623:             raise RuntimeError('Refreshed head observation expired during replanning')
624:         previous = self._policy.classify_model(attempt.label, attempt.confidence,
625:             attempt.sorting_category)
626:         current = self._policy.classify_model(fresh_attempt.label, fresh_attempt.confidence,
627:             fresh_attempt.sorting_category)
628:         if (current.category == Category.REVIEW or current.category != previous.category
629:                 or current.destination != previous.destination):
630:             raise RuntimeError('Object classification changed during head reference refresh')
631:         return refreshed, fresh_attempt

702:     def _verify_head_redetection(self, attempt, *, minimum_center_z_m=None):
703:         """Associate a new YOLOE mask by measured 3D position, not Gemini wording."""
704:         held = self._held_object
705:         if held is None:
706:             raise RuntimeError('No attachment reference for head RGB-D reobservation')
707:         minimum = float(self.get_parameter('lift_min_center_z_m').value)
708:         if minimum_center_z_m is not None:
709:             minimum = max(minimum, float(minimum_center_z_m))
710:         tolerance = float(self.get_parameter('sorting_held_association_tolerance_m').value)
711:         if not math.isfinite(minimum) or not math.isfinite(tolerance) or tolerance <= 0:
712:             raise ValueError('Invalid lift height or association tolerance')
713:         arm = held.selected.selected_arm
714:         tcp = self._tcp_pose(arm)
715:         predicted = np.array(self._pose_position(tcp)) + rotation(tcp) @ held.offset_in_tcp
716:         detected = self._detect_objects()
717:         matches = []
718:         for item in detected.detections.detections:
719:             if not item.distance_valid:
720:                 continue
721:             candidate = ObjectAttempt(
722:                 object_id=int(item.object_id), label=item.label,
723:                 confidence=float(item.confidence), distance_m=float(item.distance_m))
724:             inspected = self._inspect_selected(detected.detections.snapshot_id, candidate)
725:             if inspected is None:
726:                 continue

728:             position = np.array((center.x, center.y, center.z))
729:             error = float(np.linalg.norm(predicted - position))
730:             if math.isfinite(error) and error <= tolerance:
731:                 matches.append((inspected, error))
732:         if not matches:
733:             raise LiftRedetectionError(
734:                 f'Head RGB-D could not associate lifted {attempt.label} with the gripper')
735:         if len(matches) != 1:
736:             raise RuntimeError('Head RGB-D found multiple objects at the held-object position')
737:         inspected, error = matches[0]
738:         center_z = float(inspected.objects.objects[0].obb_pose.position.z)
739:         self.get_logger().debug(
740:             f'Head RGB-D lift verification: source={attempt.label} '
741:             f'observed={inspected.objects.objects[0].label} '
742:             f'center_z={center_z:.3f}m association_error={error:.3f}m')
743:         if not math.isfinite(center_z) or center_z < minimum:
744:             raise RuntimeError(
745:                 f'{attempt.label} was not retained after lift: '
746:                 f'center_z={center_z:.3f}m minimum={minimum:.3f}m')

751:                             tolerance_m: float) -> JointState | None:
752:         arm = held.selected.selected_arm
753:         pose = self._tcp_pose(arm)
754:         self._transport_adapter.set_current_state(self._feedback_state())

756:         for _ in range(5):

759:                 arm, tuple(float(v) for v in target), solution)

782:     def _confirm_held_contact(self, held: HeldObject) -> None:
783:         """Require consecutive post-request motor feedback while the arm is held."""
784:         duration = float(self.get_parameter('sorting_wrist_contact_confirmation_sec').value)
785:         timeout = float(self.get_parameter('gripper_contact_feedback_timeout_sec').value)
786:         maximum_age = float(self.get_parameter('sorting_gripper_feedback_max_age_sec').value)
787:         if (not all(math.isfinite(v) and v > 0 for v in (duration, timeout, maximum_age))
788:                 or duration > timeout):
789:             raise ValueError('Invalid held-contact confirmation timing')
790:         joint = f'{held.selected.selected_arm}_gripper_joint'
791:         previous = self.get_clock().now().nanoseconds
792:         first = None
793:         samples = 0
794:         deadline = time.monotonic() + timeout
795:         while time.monotonic() < deadline:
796:             self._spin_once(timeout_sec=0.05)
797:             stamp = self._gripper_feedback_stamps.get(joint, 0)
798:             if stamp <= previous:
799:                 continue
800:             now = self.get_clock().now().nanoseconds
801:             # /joint_states and /clock have separate subscriptions. A joint
802:             # update can arrive first; leave it uncounted until the clock catches
803:             # up. A stopped clock or persistently future stamp still times out.
804:             if stamp > now:
805:                 continue
806:             if (now - stamp) / 1e9 > maximum_age:
807:                 raise RuntimeError('Held-gripper feedback is stale')
808:             if first is not None and (stamp - previous) / 1e9 > maximum_age:
809:                 raise RuntimeError('Held-gripper feedback continuity was lost')
810:             self._require_held_contact(held)
811:             previous = stamp
812:             first = stamp if first is None else first
813:             samples += 1
814:             if samples >= 3 and (stamp - first) / 1e9 >= duration:
815:                 return
816:         raise RuntimeError('No fresh sustained held-gripper feedback before timeout')

901:         self._transport_adapter.set_current_state(self._feedback_state())
902:         solution = self._fixed_release_cache.get(key)
903:         if solution is not None:
904:             joints = JointState(name=list(solution.names), position=list(solution.positions))
905:             pose = self._tcp_pose(arm, joints)
906:             center = np.array(self._pose_position(pose))+rotation(pose)@offset
907:             values = dict(zip(solution.names, solution.positions, strict=True))
908:             if (np.max(np.abs(center-point)) <= .001
909:                     and all(values.get(name) == value for name, value in wrist.items())
910:                     and self._transport_adapter.state_is_valid(arm, solution)):
911:                 self.get_logger().debug('Reusing fixed release IK after fresh FK/collision checks')
912:                 return solution
913:             self._fixed_release_cache.pop(key)
914:         solution = self._transport_adapter.solve_held_region_ik(
915:             arm, point-.0005, point+.0005, offset, fixed_joints=wrist,
916:             attempts=int(self.get_parameter('sorting_release_ik_attempts').value),
917:             iterations=int(self.get_parameter('sorting_release_ik_iterations').value))
918:         if solution is not None:
919:             if len(self._fixed_release_cache) >= 64:
920:                 self._fixed_release_cache.pop(next(iter(self._fixed_release_cache)))
921:             self._fixed_release_cache[key] = solution
922:         return solution

924:     def _transport_fixed_release(self, held, destination):
925:         arm = held.selected.selected_arm
926:         point = self._fixed_release_points[destination]
927:         # Classification already selected the destination before pick. Hold the
928:         # post-lift wrist and plan directly to that bin, with no shared waypoint.
929:         self._wait_arm_stationary(arm)
930:         self._require_held_contact(held)
931:         feedback = self._arm_joint_state(arm)
932:         actual = dict(zip(feedback.name, feedback.position, strict=True))
933:         self._fixed_release_wrist = {name: actual[name] for name in self._carry_wrist_reference}
934:         drop = self._fixed_release_ik(arm, point, held.offset_in_tcp, self._fixed_release_wrist)
935:         if drop is None:
936:             raise RuntimeError(f'Direct fixed release unreachable with post-lift wrist: {arm}/{destination}')
937:         self.get_logger().info(f'Fixed release: {destination} object_center={point.tolist()} wrist held')
938:         joints = JointState(name=list(drop.names), position=list(drop.positions))
939:         self._move_to(arm, joints, f'transport to {destination}')
940:         self._verify_feedback(joints)
941:         self._require_held_contact(held)

949:     def check_release_target(self, held: HeldObject, destination: str) -> None:
950:         arm = held.selected.selected_arm
951:         pose = self._tcp_pose(arm)
952:         center = (np.array(self._pose_position(pose))
953:                   + rotation(pose) @ held.offset_in_tcp)
954:         if not np.isfinite(center).all():
955:             raise RuntimeError('Release object center must be finite')
956:         if getattr(self, '_fixed_release_wrist', {}):
957:             if np.max(np.abs(center-self._fixed_release_points[destination])) > .01:
958:                 raise RuntimeError('Fixed release object center has not reached its destination')
959:             state = self._arm_joint_state(arm)
960:             feedback = dict(zip(state.name, state.position, strict=True))
961:             if any(name not in feedback or not math.isfinite(feedback[name])
962:                    or abs(feedback[name]-value) > self._fixed_release_hold_tolerance+1e-6
963:                    for name, value in self._fixed_release_wrist.items()):
964:                 raise RuntimeError('Fixed release wrist feedback is outside hold tolerance')
965:         # TODO(bin-opening-fit): legacy automatic sorting still defers this check.
966:         # The single-object Action adds the check in MujocoOperations.

1008:     def retreat(self, held: HeldObject, destination: str, *, prefetch: bool = True):
1009:         arm = held.selected.selected_arm
1010:         if self._held_object is not None:
1011:             raise RuntimeError('Cannot close the returning gripper before release completes')
1012:         joints = deepcopy(self._home[arm])
1013:         closing = float(self.get_parameter('sorting_return_gripper_position_rad').value)
1014:         if not math.isfinite(closing):
1015:             raise ValueError('Return gripper position must be finite')
1016:         joints.name.append(f'{arm}_gripper_joint')
1017:         joints.position.append(closing)
1018:         joints.velocity, joints.effort = [], []
1019:         if self._wrist_enabled:
1020:             self._switch_camera('head')
1021:         # InspectScene waits for a new RGB-D frame after accepting the goal.
1022:         # Inference proceeds in the perception node while this node executes return.
1023:         if prefetch:
1024:             self._return_detection = self._begin_object_detection()
1025:             self.get_logger().info('Next scene detection submitted before return motion')
1026:         self.get_logger().info(f'Return with simultaneous {arm} gripper closing')
1027:         try:
1028:             self._move_to(arm, joints, f'return from {destination}')
1029:             self._verify_feedback(joints)
1030:         except Exception:
1031:             if self._return_detection is not None:
1032:                 self._return_detection[0].cancel_goal_async()
1033:                 self._return_detection = None
1034:             raise

1057:     def _observe_wrist(self, selected, operation, *, held=False, after_stamp_ns=None):
1058:         candidate = selected.selected_candidate
1059:         obj = candidate.target_object
1060:         request = ObserveWristTarget.Request(operation=operation, arm=selected.selected_arm,
1061:             source_snapshot_id=candidate.snapshot_id, source_object_id=obj.object_id,
1062:             label=obj.label, source_confidence=obj.confidence,
1063:             detector_label=obj.detector_label,
1064:             reference_id=self._wrist_reference.reference_id if self._wrist_reference else '',
1065:             size=deepcopy(obj.obb_size), after_stamp_ns=self.get_clock().now().nanoseconds)
1066:         if operation == ObserveWristTarget.Request.CHECK:
1067:             completed = getattr(self, '_lift_completion_stamp_ns', None)
1068:             if completed is None or not 0 < completed <= request.after_stamp_ns:
1069:                 raise RuntimeError('Missing or invalid lift completion freshness barrier')
1070:             # Results captured during the existing stabilization hold are valid
1071:             # post-lift evidence. Do not force another full frame after the hold.
1072:             request.after_stamp_ns = max(completed, after_stamp_ns or 0)
1073:         request.expected_pose.header=deepcopy(candidate.header)
1074:         request.expected_pose.pose=deepcopy(obj.obb_pose)
1075:         if held:
1076:             tcp = self._tcp_pose(selected.selected_arm)
1077:             center = np.array(self._pose_position(tcp))+rotation(tcp)@self._held_object.offset_in_tcp
1078:             request.expected_pose.pose.position.x, request.expected_pose.pose.position.y, request.expected_pose.pose.position.z = map(float,center)
1079:             from cleany_skill_executor.core.wrist_pose import carried_orientation
1080:             request.expected_pose.pose.orientation = carried_orientation(
1081:                 self._contact_tcp_quaternion, tcp.orientation, obj.obb_pose.orientation)
1082:         self._record_pipeline_message('wrist_request', request)
1083:         started = time.monotonic()
1084:         response = self._future(self._wrist_client.call_async(request), 30., 'wrist RGB verification')
1085:         self.get_logger().debug(f'WRIST RPC operation={operation} elapsed_sec={time.monotonic()-started:.3f} '
1086:                                f'after_ns={request.after_stamp_ns}')
1087:         self._record_pipeline_message('wrist_response', response)
1088:         missing = (not response.success and response.status == response.NOT_DETECTED
1089:                    and operation == ObserveWristTarget.Request.CHECK)
1090:         if not missing and (not response.success or response.status != response.OK):
1091:             raise RuntimeError(f'Wrist RGB verification failed: {response.message}')
1092:         stamp=response.header.stamp.sec*10**9+response.header.stamp.nanosec
1093:         if (response.source_snapshot_id != candidate.snapshot_id
1094:                 or response.source_object_id != obj.object_id
1095:                 or response.header.frame_id != f'{selected.selected_arm}_wrist_rgb_optical_frame'
1096:                 or not request.after_stamp_ns < stamp <= self.get_clock().now().nanoseconds
1097:                 or not response.reference_id
1098:                 or (operation==ObserveWristTarget.Request.CHECK and
1099:                     response.reference_id != request.reference_id)):
1100:             raise RuntimeError('Wrist response source/frame/timestamp mismatch')
1101:         if missing:
1102:             raise WristTargetNotDetected(response.message, stamp)
1103:         return response
