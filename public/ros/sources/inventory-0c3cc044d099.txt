195:     def _wait_future(self, future, deadline: float, goal_handle, label: str):
196:         while not future.done():
197:             if goal_handle is not None and goal_handle.is_cancel_requested:
198:                 future.cancel()
199:                 raise InterruptedError('validate_pick canceled')
200:             if time.monotonic() >= deadline:
201:                 future.cancel()
202:                 raise RuntimeError(f'{label} timed out')
203:             time.sleep(0.01)
204:         if goal_handle is not None and goal_handle.is_cancel_requested:
205:             raise InterruptedError('validate_pick canceled')
206:         if time.monotonic() >= deadline:
207:             raise RuntimeError(f'{label} timed out')
208:         try:
209:             value = future.result()
210:         except Exception as error:
211:             raise RuntimeError(f'{label} failed: {error}') from error
212:         if value is None:
213:             raise RuntimeError(f'{label} returned no result')
214:         return value

558:                     record.update(status='plan_failed', detail=str(error), error_code=error.code)
559:                     # A failed clear stops the request; do not silently leak a pinned target.
560:                     try:
561:                         self._clear_reference(result)
562:                     except RuntimeError as cleanup_error:
563:                         record.update(status='interrupted', detail=str(cleanup_error),
564:                                       error_code=result.ERROR_PERCEPTION)
565:                         raise _StageFailure(result.ERROR_PERCEPTION, str(cleanup_error)) from cleanup_error
566:                     self.get_logger().info(
567:                         f'Object {selected_detection.object_id} skipped: {error}')
568:                     self._feedback(goal_handle, ValidatePick.Feedback.STAGE_FILTERING,
569:                         f'Object {selected_detection.object_id}: {error}; trying next object',
570:                         detection_count=len(phase1.detections.detections))
571:                     latencies.extend((f'object_{selected_detection.object_id}/{name}', seconds)
572:                                      for name, seconds in object_latencies)
573:                     continue
574:                 except Exception as error:
575:                     record.update(status='interrupted', detail=str(error), error_code=(
576:                         error.code if isinstance(error, _StageFailure) else
577:                         result.ERROR_CANCELED if isinstance(error, InterruptedError) else result.ERROR_INTERNAL))
578:                     latencies.extend((f'object_{selected_detection.object_id}/{name}', seconds)
579:                                      for name, seconds in object_latencies)
580:                     raise
581:                 else:
582:                     record.update(status='selected', detail='Validated grasp and motion paths')
583:                     latencies.extend(object_latencies)
584:                     break
585:                 finally:
586:                     record['elapsed_sec'] = time.monotonic() - started_object
587:                     record['latency_seconds'] = dict(object_latencies)
588:                     record['candidate_count'] = len(result.candidates)
589:             else:
590:                 assert last_failure is not None
591:                 raise _StageFailure(last_failure.code,
592:                     'No validated grasp for any eligible object: ' + '; '.join(
593:                         f"{item['object_id']}:{item['detail']}" for item in attempts
594:                         if item['status'] == 'plan_failed'))
595:             result.latency_stages = [name for name, _ in latencies]
596:             result.latency_seconds = [seconds for _, seconds in latencies]
597: 
598:             evidence_root = str(
599:                 self.get_parameter('evidence_output_directory').value
600:             ).strip()
601:             if evidence_root:
602:                 self._feedback(
603:                     goal_handle,
604:                     ValidatePick.Feedback.STAGE_RECORDING,
605:                     'Writing JSON and 3x3 evidence',
606:                     detection_count=len(phase1.detections.detections),
607:                     candidate_count=len(result.candidates),
608:                 )
609:                 _, grasp_debug = self._debug_messages()
610:                 try:
611:                     json_path, grid_path = write_validation_evidence(
612:                         Path(evidence_root),
613:                         snapshot_id=phase1.detections.snapshot_id,
614:                         query=goal_handle.request.query,
615:                         detections=phase1.detections.detections,
616:                         decisions=decisions,
617:                         selected_detection=selected_detection,
618:                         selected_object=phase2.objects.objects[0],
619:                         candidates=result.candidates,
620:                         selection=selected,
621:                         latencies=latencies,
622:                         phase1_debug=phase1_debug,
623:                         phase2_debug=phase2_debug,
624:                         grasp_debug=grasp_debug,
625:                         target_attempts=attempts,
626:                     )
627:                     result.evidence_json_path = json_path
628:                     result.evidence_grid_path = grid_path
629:                 except Exception as error:
630:                     self.get_logger().error(f'Failed to write evidence: {error}')
631:             result.success = True
632:             result.error_code = result.ERROR_NONE
633:             result.message = (
634:                 f'Validated AnyGrasp candidate {result.selected_candidate_index} '
635:                 f'with {result.selected_arm} arm (plan-only)'
636:             )
637:             terminal = 'succeed'
638:         except InterruptedError as error:
639:             self._set_failure(result, result.ERROR_CANCELED, str(error))
640:             terminal = 'canceled'
641:         except _StageFailure as error:
642:             self._set_failure(result, error.code, str(error))
643:         except Exception as error:
644:             self._set_failure(
645:                 result, result.ERROR_INTERNAL, f'Unexpected validation error: {error}'
646:             )
647:         finally:
648:             self._active_downstream_goal = None
649:             if terminal != 'succeed' and result.reference_id:
650:                 try:
651:                     self._clear_reference(result)
652:                 except RuntimeError as error:
653:                     self.get_logger().error(str(error))
654:                     self._set_failure(result, result.ERROR_PERCEPTION,
655:                         f'{result.message}; reference cleanup failed: {error}')
656:                     terminal = 'abort'
657:             result.latency_stages = [name for name, _ in latencies]
658:             result.latency_seconds = [seconds for _, seconds in latencies]
659:             for record in attempts:
660:                 if record['status'] == 'not_checked':
661:                     record['detail'] = (f'Object {result.selected_detection.object_id} selected first'
662:                         if result.success else f'Request stopped: {result.message}')
663:             self.get_logger().info('Target attempts: ' + json.dumps(attempts, ensure_ascii=False))
664:             evidence_root = str(self.get_parameter('evidence_output_directory').value).strip()
665:             if evidence_root and snapshot_id:
666:                 try:
667:                     attempts_path = write_target_attempt_evidence(Path(evidence_root),
668:                         snapshot_id=snapshot_id, attempts=attempts,
669:                         success=result.success, error_code=result.error_code, message=result.message)
670:                     if not result.evidence_json_path:
671:                         result.evidence_json_path = attempts_path
672:                 except Exception as error:
673:                     self.get_logger().error(f'Failed to write target attempts: {error}')
674:             with self._goal_lock:
675:                 self._goal_active = False
676:         if terminal == 'succeed':
677:             goal_handle.succeed()
678:         elif terminal == 'canceled':
679:             goal_handle.canceled()
680:         else:
681:             goal_handle.abort()
682:         return result
