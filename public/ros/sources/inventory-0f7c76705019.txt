53: class MujocoSimNode(Node):
54:     """Step the standalone MuJoCo model and expose base feedback.
55: 
56:     Subscriptions:
57:         ~/joint_cmd (sensor_msgs/msg/JointState): Named joint targets.
58:         cmd_vel (geometry_msgs/msg/Twist): Bounded planar base velocity.
59: 
60:     Publishers:
61:         joint_states (sensor_msgs/msg/JointState): Simulated joint state.
62:         odom (nav_msgs/msg/Odometry): Base pose and velocity.
63:         scan (sensor_msgs/msg/LaserScan): Ray-cast range data when scan_enabled.
64: 
65:     TF:
66:         Optionally broadcasts odom -> base_link and the static LiDAR mount.
67:         odom, base_link and laser are configurable frame defaults.
68: 
69:     Notes:
70:         Requires a valid scene_path. Wheel dynamics, command limits and timeout
71:         are simulation settings; this node is separate from ros2_control's backend.
72:     """

291:     def _on_cmd_vel(self, msg: Twist) -> None:
292:         all_axes = (
293:             msg.linear.x,
294:             msg.linear.y,
295:             msg.linear.z,
296:             msg.angular.x,
297:             msg.angular.y,
298:             msg.angular.z,
299:         )
300:         if not are_finite_values(all_axes):
301:             self.get_logger().warning(
302:                 'Ignoring non-finite cmd_vel and stopping the base'
303:             )
304:             self._stop_chassis_command()
305:             return
306: 
307:         unsupported_axes = (msg.linear.z, msg.angular.x, msg.angular.y)
308:         if any(value != 0.0 for value in unsupported_axes):
309:             self.get_logger().warning(
310:                 'Ignoring unsupported cmd_vel axes; only linear.x, linear.y, '
311:                 'and angular.z are used'
312:             )
313: 
314:         try:
315:             chassis_command = bounded_command(
316:                 ChassisCommand(
317:                     linear_x=msg.linear.x,
318:                     linear_y=msg.linear.y,
319:                     angular_z=msg.angular.z,
320:                 ),
321:                 self._command_limits,
322:             )
323:             target_wheel_speeds = limited_wheel_speeds(
324:                 wheel_speeds_from_chassis(
325:                     chassis_command,
326:                     self._mecanum_geometry,
327:                 ),
328:                 self._wheel_speed_limit,
329:             )
330:         except ValueError:
331:             self.get_logger().warning(
332:                 'Ignoring invalid cmd_vel and stopping the base'
333:             )
334:             self._stop_chassis_command()
335:             return
336: 
337:         self._current_chassis_command = chassis_command
338:         self._target_wheel_speeds = target_wheel_speeds
339:         self._last_cmd_vel_time = time.monotonic()
340: 
341:     def _on_cmd_vel_timeout(self) -> None:
342:         if self._last_cmd_vel_time is None:
343:             return
344:         if time.monotonic() - self._last_cmd_vel_time <= self._cmd_vel_timeout_sec:
345:             return
346: 
347:         self.get_logger().warning('cmd_vel timed out; stopping the base')
348:         self._stop_chassis_command()
349: 
350:     def _stop_chassis_command(self) -> None:
351:         self._current_chassis_command = stopped_command()
352:         self._target_wheel_speeds = stopped_wheel_speeds()
353:         self._last_cmd_vel_time = None
354:         if self._mujoco_drive is not None:
355:             self._mujoco_drive.reset()
356: 
357:     def _on_timer(self) -> None:
358:         for _ in range(self._steps_per_tick):
359:             if self._mujoco_drive is not None:
360:                 self._mujoco_drive.apply_control(
361:                     self._data,
362:                     self._target_wheel_speeds,
363:                     self._model.opt.timestep,
364:                 )
365:             mujoco.mj_step(self._model, self._data)
366:         stamp = self.get_clock().now()
367:         for observer in self._step_observers:
368:             observer.after_step(self._simulation_context, stamp)
369:         self._joint_state_pub.publish(joint_state_msg(self._model, self._data, stamp))
370:         self._odom_pub.publish(
371:             odometry_msg(
372:                 self._model,
373:                 self._data,
374:                 self._base_body_id,
375:                 stamp,
376:                 self._odom_frame_id,
377:                 self._base_frame_id,
378:             )
379:         )
380:         if self._publish_odom_tf:
381:             self._tf_broadcaster.sendTransform(
382:                 transform_msg(
383:                     self._data,
384:                     self._base_body_id,
385:                     stamp,
386:                     self._odom_frame_id,
387:                     self._base_frame_id,
388:                 )
389:             )
390:         should_publish_scan = (
391:             self._scan_enabled
392:             and self._data.time - self._sim_time_at_last_scan >= 1.0 / self._scan_rate_hz
393:         )
394:         if should_publish_scan:
395:             self._scan_pub.publish(
396:                 laser_scan_msg(
397:                     self._model,
398:                     self._data,
399:                     self._lidar_site_id,
400:                     self._base_body_id,
401:                     stamp,
402:                     self._laser_frame_id,
403:                     self._scan_rate_hz,
404:                     self._scan_samples,
405:                     self._scan_range_min,
406:                     self._scan_range_max,
407:                 )
408:             )
409:             self._sim_time_at_last_scan = self._data.time
410:         if self._viewer is not None and time.monotonic() >= self._next_viewer_sync:
411:             self._viewer.sync()
412:             self._next_viewer_sync = time.monotonic() + 1.0 / self._viewer_rate_hz
