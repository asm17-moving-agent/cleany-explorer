34:     def inputs_ready(self):
35:         joints = self.joints
36:         if (joints is None or len(self.infos) != 4
37:                 or len(joints.name) != len(joints.position)
38:                 or not set(REQUIRED_JOINT_NAMES) <= set(joints.name)
39:                 or not all(math.isfinite(q) for q in joints.position)):
40:             self.last_reason = 'Missing camera info or complete measured joints'
41:             return False
42:         now = self.get_clock().now().nanoseconds
43:         for message in (joints, *self.infos.values()):
44:             stamp = Time.from_msg(message.header.stamp)
45:             if stamp.nanoseconds <= 0 or not 0 <= now-stamp.nanoseconds <= 500_000_000:
46:                 self.last_reason = f'Stale/future input: {message.header.frame_id}'
47:                 return False
48:             if isinstance(message, CameraInfo) and not self.tf.can_transform(
49:                     'base_link', message.header.frame_id, stamp):
50:                 self.last_reason = f'Missing capture-time TF: base_link <- {message.header.frame_id}'
51:                 return False
52:         self.last_reason = 'Waiting for paired head camera info and active controllers'
53:         return (self.infos['/camera/color/camera_info'].header.stamp
54:                 == self.infos['/camera/depth/camera_info'].header.stamp)
