37:         if not 0 < rate <= 50:
38:             raise ValueError('rate_hz must be in (0, 50]')

61:     def poll(self):
62:         # Motors are sampled sequentially; this timestamp is the start of the read.
63:         stamp = self.get_clock().now().to_msg()
64:         try:
65:             positions = [joint.radians(self.arm.read(joint.motor_id, 56, 2))
66:                          for joint in self.joints]
67:             positions += head_positions([self.head.read(i, 56, 2) for i in (13, 14)],
68:                                         self.head_mapping)
69:         except (RuntimeError, ValueError, OSError) as error:
70:             if time.monotonic() - self.last_warning > 2:
71:                 self.get_logger().error(f'No complete fresh state published: {error}')
72:                 self.last_warning = time.monotonic()
73:             return
74:         message = JointState()
75:         message.header.stamp = stamp
76:         message.name = [joint.name for joint in self.joints] + ['head_pan_joint', 'head_tilt_joint']
77:         message.position = positions
78:         # Velocity/effort are absent: no unverified values are manufactured.
79:         self.publisher.publish(message)
