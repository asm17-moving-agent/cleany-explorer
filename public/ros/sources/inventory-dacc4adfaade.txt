90:     def _tick(self) -> None:
91:         with self.core.lock:
92:             if self._closing:
93:                 return
94:             if self.core.record is not None:
95:                 super()._tick()
96:             if self.core.tree is not None:
97:                 execution = self.core.record.goal.execution_id
98:                 if execution != self._monitor_failed_execution:
99:                     try:
100:                         self.monitor.update(self.core.tree.snapshot(), new_execution=execution != self._monitor_execution)
101:                     except Exception as error:
102:                         self._monitor_failed_execution = execution
103:                         self.get_logger().warning(f'BT monitor unavailable for {execution}: {error}')
104:                     else:
105:                         self._monitor_execution = execution
106:                 for event in self.core.transitions:
107:                     self._bt_events.publish(String(data=json.dumps(dict(execution_id=execution, **event))))
108:                 self.core.transitions.clear()
109:             for _ in range(10):
110:                 try:
111:                     frames = self.socket.recv_multipart(flags=zmq.NOBLOCK)
112:                 except zmq.Again:
113:                     break
114:                 self.socket.send_multipart(self.monitor.reply(frames))
115: 
116:     def destroy_node(self):
117:         with self.core.lock:
118:             self._closing = True
119:             self._timer.cancel()
120:             self._action.destroy()
121:             # Shutdown only halts work; the ACTIVE record remains for recovery.
122:             if self.core.tree is not None and self.core.record.result is None:
123:                 self.core.tree.halt()
124:         self.backend.close()
125:         if self.operations is not None:
126:             self.operations.destroy_node()
127:         self.socket.close()
128:         self.zmq_context.term()
129:         with self.core.lock:
130:             self.store.close()
131:         return Node.destroy_node(self)
