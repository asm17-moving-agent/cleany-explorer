423:     def _candidate_execution_positions(
424:         self,
425:         candidate,
426:     ) -> tuple[np.ndarray, np.ndarray]:
427:         approach = np.asarray(
428:             (
429:                 candidate.approach_direction.x,
430:                 candidate.approach_direction.y,
431:                 candidate.approach_direction.z,
432:             ),
433:             dtype=float,
434:         )
435:         approach /= np.linalg.norm(approach)
436:         orientation = candidate.tcp_pose.orientation
437:         closing = np.asarray(
438:             quaternion_axis(
439:                 (
440:                     orientation.x,
441:                     orientation.y,
442:                     orientation.z,
443:                     orientation.w,
444:                 ),
445:                 (1.0, 0.0, 0.0),
446:             ),
447:             dtype=float,
448:         )
449:         source = np.asarray(
450:             (
451:                 candidate.tcp_pose.position.x,
452:                 candidate.tcp_pose.position.y,
453:                 candidate.tcp_pose.position.z,
454:             ),
455:             dtype=float,
456:         )
457:         lateral = float(
458:             self.get_parameter('selector_grasp_lateral_offset_m').value
459:         )
460:         if getattr(self, '_aperture_centering', False):
461:             lateral = aperture_centering_offset(float(candidate.required_opening_m),
462:                 float(self.get_parameter('grasp_aperture_margin_m').value),
463:                 float(self.get_parameter('grasp_fixed_jaw_inner_x_m').value),
464:                 float(self.get_parameter('grasp_fixed_jaw_clearance_m').value))
465:         pregrasp = (
466:             source
467:             - float(
468:                 self.get_parameter('selector_pregrasp_offset_m').value
469:             )
470:             * approach
471:             + lateral * closing
472:         )
473:         grasp = (
474:             source

636:     def _wait_for_pipeline(self) -> None:
637:         startup_timeout = float(
638:             self.get_parameter('startup_timeout_sec').value
639:         )
640:         self.get_logger().info(
641:             'Waiting for perception, grasp, reachable-grasp, and MoveGroup'
642:         )
643:         if not self._inspection.wait_for_server(timeout_sec=startup_timeout):
644:             raise RuntimeError('inspection action is unavailable')
645:         if not self._grasp.wait_for_service(timeout_sec=startup_timeout):
646:             raise RuntimeError('grasp planning service is unavailable')
647:         if not self._selection.wait_for_server(timeout_sec=startup_timeout):

649:         if not self._move_group.wait_for_server(timeout_sec=startup_timeout):
650:             raise RuntimeError('MoveGroup action is unavailable')

662:         # snapshot.

668:     def _run_nearest(self) -> None:
669:         detection_result = self._detect_objects()
670:         attempts = self._attempts(detection_result.detections.detections)
671:         if not attempts:
672:             raise RuntimeError(
673:                 'no detection has enough valid depth for autonomous selection'
674:             )
675: 
676:         failures: list[str] = []
677:         for attempt in attempts:
678:             self.get_logger().info(
679:                 f'Trying nearest object_id={attempt.object_id} '
680:                 f'label={attempt.label} distance={attempt.distance_m:.3f}m'
681:             )
682:             inspected = self._inspect_selected(
683:                 detection_result.detections.snapshot_id,
684:                 attempt,
685:             )
686:             if inspected is None:
687:                 failures.append(f'{attempt.object_id}: inspection')
688:                 continue
689:             planned = self._plan_grasps(inspected, attempt)
690:             if planned is None:
691:                 failures.append(f'{attempt.object_id}: no grasp')
692:                 continue
693:             self._publish_grasp_overlay(planned.candidates)
694:             selected = self._select_reachable(planned.candidates, attempt)
695:             if selected is None:
696:                 failures.append(f'{attempt.object_id}: unreachable')
697:                 continue
698:             self._publish_grasp_overlay(
699:                 planned.candidates,
700:                 selected_index=selected.selected_candidate_index,
701:                 selected_arm=selected.selected_arm,
702:             )
703:             if bool(self.get_parameter('plan_only').value):
704:                 self.get_logger().info(
705:                     'NEAREST PLAN COMPLETE (plan_only): no arm/gripper motion'
706:                 )
707:                 return
708:             self._execute_pregrasp(selected, attempt)
709:             if bool(self.get_parameter('execute_grasp_and_lift').value):
710:                 # Every failure before physical gripper contact leaves the
711:                 # selected gripper open at the stationary pre-grasp pose.
712:                 self._open_gripper(selected.selected_arm)
713:                 refreshed, refreshed_attempt = self._refresh_selected_grasp(
714:                     selected, attempt
715:                 )
716:                 self._execute_grasp_and_lift(
717:                     refreshed, refreshed_attempt
718:                 )
719:                 self.get_logger().info(
720:                     'NEAREST GRASP COMPLETE: '
721:                     f'object_id={attempt.object_id} label={attempt.label} '
722:                     f'distance={attempt.distance_m:.3f}m '
723:                     f'candidate={refreshed.selected_candidate_index} '
724:                     f'arm={refreshed.selected_arm}'
725:                 )
726:                 return
727:             self.get_logger().info(
728:                 'NEAREST PREGRASP COMPLETE: '
729:                 f'object_id={attempt.object_id} label={attempt.label} '
730:                 f'distance={attempt.distance_m:.3f}m '
731:                 f'candidate={selected.selected_candidate_index} '
732:                 f'arm={selected.selected_arm}'
733:             )
734:             return
735:         raise RuntimeError(
736:             'no distance-valid object produced a reachable pre-grasp; '
737:             + ', '.join(failures)
738:         )

754:     def _check_sensor_scene(self) -> None:
755:         # Humble's service serialization can race live OctoMap writes. The
756:         # scene publisher takes the OctoMap read lock: consume its output.
757:         age = float('inf') if self._scene_cloud_stamp_ns is None else (
758:             self.get_clock().now().nanoseconds - self._scene_cloud_stamp_ns
759:         ) / 1e9
760:         # The processed-cloud receipt is published after tree writes. A static
761:         # populated map need not be serialized again to prove a new integration.
762:         # Raw cloud arrival alone must never renew this evidence.
763:         map_age = time.monotonic() - max(self._sensor_map_received_at,
764:             getattr(self, '_scene_commit_received_at', float('-inf')))
765:         maximum_age = float(self.get_parameter('sensor_scene_maximum_age_sec').value)
766:         try:
767:             validate_sensor_scene(
768:                 max(age, map_age) if age >= 0 else age,
769:                 maximum_age, self._sensor_map_id, self._sensor_map_bytes,
770:             )
771:         except RuntimeError as error:
772:             raise RuntimeError(
773:                 f'{error} (cloud_age={age:.3f}s map_receipt_age={map_age:.3f}s '
774:                 f'maximum={maximum_age:.3f}s)'
775:             ) from error

784:     def _wait_for_sensor_scene(self, timeout_sec: float, *, after_stamp_ns: int | None = None) -> None:
785:         self.get_logger().info('Waiting for fresh depth and populated OctoMap')
786:         started = time.monotonic()
787:         deadline = started + timeout_sec
788:         while True:
789:             try:
790:                 self._check_sensor_scene()
791:                 if after_stamp_ns is not None and (
792:                         self._scene_cloud_stamp_ns is None or self._scene_cloud_stamp_ns <= after_stamp_ns):
793:                     raise RuntimeError('No processed depth capture after the attachment scene update')
794:                 if after_stamp_ns is not None:
795:                     self.get_logger().debug(
796:                         f'Post-attachment depth ready: wait_wall_sec={time.monotonic()-started:.3f} '
797:                         f'capture_after_attachment_sec={(self._scene_cloud_stamp_ns-after_stamp_ns)/1e9:.3f}')
798:                 return
799:             except RuntimeError:
800:                 if time.monotonic() >= deadline:
801:                     raise
802:                 self._spin_once(timeout_sec=0.2)

1390:     @staticmethod
1391:     def _obb_corners(box) -> np.ndarray:
1392:         size = box.obb_size
1393:         pose = box.obb_pose
1394:         extents = np.array((size.x, size.y, size.z))
1395:         center = np.array((pose.position.x, pose.position.y, pose.position.z))
1396:         if not np.isfinite(extents).all() or np.any(extents <= 0) or not np.isfinite(center).all():
1397:             raise ValueError('OBB requires finite center and positive extents')
1398:         q = pose.orientation
1399:         rotation = rotation_matrix_from_quaternion(q.x, q.y, q.z, q.w)
1400:         signs = np.array([[x, y, z] for x in (-0.5, 0.5)
1401:                           for y in (-0.5, 0.5) for z in (-0.5, 0.5)])
1402:         return (signs * extents) @ rotation.T + center

1799:     def _contact_sample(
1800:         self, arm: str, target: Pose
1801:     ) -> ContactSample | None:
1802:         state = self._controller_states.get(arm)
1803:         if state is None:
1804:             return None
1805:         if (
1806:             len(state.reference.positions) == len(state.feedback.positions)
1807:             and state.reference.positions
1808:         ):
1809:             errors = tuple(
1810:                 float(reference) - float(feedback)
1811:                 for reference, feedback in zip(
1812:                     state.reference.positions,
1813:                     state.feedback.positions,
1814:                     strict=True,
1815:                 )
1816:             )
1817:         else:
1818:             errors = tuple(float(value) for value in state.error.positions)
1819:         velocities = tuple(float(value) for value in state.feedback.velocities)
1820:         if not errors or not velocities:
1821:             return None
1822:         effort = (
1823:             tuple(float(value) for value in state.feedback.effort)
1824:             if state.feedback.effort
1825:             else None
1826:         )
1827:         # FK is queried only for a plausible stopped/error sample.
1828:         distance = math.inf
1829:         effort_threshold = float(
1830:             self.get_parameter('contact_effort_threshold').value
1831:         )
1832:         plausible_signal = (
1833:             max(abs(value) for value in errors)
1834:             >= float(
1835:                 self.get_parameter('contact_min_joint_error_rad').value
1836:             )
1837:             or (
1838:                 effort_threshold > 0.0
1839:                 and effort is not None
1840:                 and max(abs(value) for value in effort) >= effort_threshold
1841:             )
1842:         )
1843:         if plausible_signal and max(abs(value) for value in velocities) <= float(
1844:             self.get_parameter(
1845:                 'contact_max_joint_velocity_rad_s'
1846:             ).value
1847:         ):
1848:             distance = math.dist(
1849:                 self._pose_position(self._tcp_pose(arm)),
1850:                 self._pose_position(target),
1851:             )
1852:         return ContactSample(distance, errors, velocities, effort)

1995:     def _wait_for_gripper_contact(self, arm: str, start: float, command: float) -> bool:
1996:         """Observe settling after close without sending another motor command."""
1997:         timeout = float(self.get_parameter('gripper_contact_feedback_timeout_sec').value)
1998:         duration = float(self.get_parameter('gripper_contact_stable_duration_sec').value)
1999:         if not all(math.isfinite(v) and v > 0 for v in (timeout, duration)):
2000:             raise ValueError('Gripper contact feedback durations must be positive and finite')
2001:         joint = f'{arm}_gripper_joint'
2002:         started = time.monotonic()
2003:         previous = getattr(self, '_gripper_feedback_stamps', {}).get(joint, 0)
2004:         stable_since = None
2005:         samples = 0
2006:         while time.monotonic() - started < timeout:
2007:             self._spin_once(timeout_sec=0.05)
2008:             stamp = getattr(self, '_gripper_feedback_stamps', {}).get(joint, 0)
2009:             if stamp <= previous:
2010:                 continue
2011:             previous = stamp
2012:             if self._gripper_contact_stalled(arm, start, command):
2013:                 if stable_since is None:
2014:                     stable_since = stamp
2015:                 samples += 1
2016:                 if samples >= 3 and (stamp - stable_since) / 1e9 >= duration:
2017:                     self.get_logger().debug(
2018:                         f'{arm} gripper stable contact: samples={samples} '
2019:                         f'wait_wall_sec={time.monotonic()-started:.3f}')
2020:                     return True
2021:             else:
2022:                 stable_since = None
2023:                 samples = 0
2024:                 actual = self._joint_positions.get(joint, math.inf)
2025:                 velocity = self._joint_velocities.get(joint, math.inf)
2026:                 limit = float(self.get_parameter('gripper_contact_max_velocity_rad_s').value)
2027:                 if abs(actual-command) <= .05 and abs(velocity) <= limit:
2028:                     return False  # Fully closed, with no object contact.
2029:         raise RuntimeError(f'{arm} gripper contact did not stabilize within {timeout:.1f}s')
