15: class ReadOnlyBus:
16:     def __init__(self, port: str, baudrate: int):
17:         from scservo_sdk import PacketHandler, PortHandler
18:         self.port = PortHandler(port)
19:         self.packet = PacketHandler(0)
20:         self.exclusive = False
21:         try:
22:             if not self.port.setBaudRate(baudrate):
23:                 raise RuntimeError('Cannot open motor port')
24:             fcntl.flock(self.port.ser.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
25:             # Prevent subsequent ordinary opens of this serial device.
26:             import termios
27:             fcntl.ioctl(self.port.ser.fileno(), termios.TIOCEXCL)
28:             self.exclusive = True
29:         except Exception:
30:             self.close()
31:             raise
32: 
33:     def read(self, motor: int, address: int, size: int) -> int:
34:         method = self.packet.read1ByteTxRx if size == 1 else self.packet.read2ByteTxRx
35:         value, status, error = method(self.port, motor, address)
36:         if status != 0 or error != 0:
37:             raise RuntimeError(f'ID {motor}: read failed comm={status} error={error}')
38:         return value
39: 
40:     def close(self):
41:         serial = getattr(self.port, 'ser', None)
42:         if serial is not None and serial.is_open:
43:             from contextlib import suppress
44:             import termios
45:             if self.exclusive:
46:                 with suppress(OSError):
47:                     fcntl.ioctl(serial.fileno(), termios.TIOCNXCL)
48:             self.exclusive = False
49:             self.port.closePort()

52: def verify_arm_calibration(bus, joints):
53:     for joint in joints:
54:         if bus.read(joint.motor_id, 3, 2) != joint.model:
55:             raise ValueError(f'{joint.name}: unexpected motor model')
56:         homing = bus.read(joint.motor_id, 31, 2)
57:         homing = -(homing & 0x7ff) if homing & 0x800 else homing
58:         actual = (homing, bus.read(joint.motor_id, 9, 2), bus.read(joint.motor_id, 11, 2))
59:         if actual != (joint.homing_offset, joint.range_min, joint.range_max):
60:             raise ValueError(f'{joint.name}: EEPROM calibration mismatch')

63: class ArmStateReader(Node):
64:     def __init__(self):
65:         super().__init__('arm_state_reader')
66:         self.bus = None
67:         self.declare_parameter('config_file', '')
68:         config = yaml.safe_load(Path(self.get_parameter('config_file').value).read_text())
69:         directory = Path(config['calibration_dir']).expanduser()
70:         calibrations = {
71:             side: json.loads((directory / arm['calibration']).read_text())
72:             for side, arm in config['arms'].items()
73:         }
74:         self.joints = mappings(config, calibrations)
75:         rate = float(config['rate_hz'])
76:         if not 0 < rate <= 50:
77:             raise ValueError('rate_hz must be in (0, 50]')
78:         self.publisher = self.create_publisher(JointState, 'joint_states', 10)
79:         try:
80:             self.bus = ReadOnlyBus(config['port'], int(config['baudrate']))
81:             verify_arm_calibration(self.bus, self.joints)
82:             self.get_logger().warning(
83:                 'READ ONLY visualization: URDF zero/direction UNVERIFIED. '
84:                 'No torque or goal writes. Do not use for motion planning.')
85:             self.last_warning = 0.0
86:             self.timer = self.create_timer(1.0 / rate, self.poll)
87:         except Exception:
88:             if self.bus:
89:                 self.bus.close()
90:             raise
91: 
92:     def poll(self):
93:         stamp = self.get_clock().now().to_msg()
94:         try:
95:             positions = [j.radians(self.bus.read(j.motor_id, 56, 2)) for j in self.joints]
96:         except (RuntimeError, ValueError, OSError) as error:
97:             if time.monotonic() - self.last_warning > 2:
98:                 self.get_logger().error(f'No fresh state published: {error}')
99:                 self.last_warning = time.monotonic()
100:             return
101:         message = JointState()
102:         message.header.stamp = stamp
103:         message.name = [j.name for j in self.joints]
104:         message.position = positions
105:         self.publisher.publish(message)
106: 
107:     def destroy_node(self):
108:         if self.bus:
109:             self.bus.close()
110:         return super().destroy_node()
