807:         result = InspectScene.Result()
808:         result.detections = self._empty_detections()
809:         result.objects = self._empty_objects()

811:             has_snapshot = bool(goal_handle.request.snapshot_id)
812:             has_selection = goal_handle.request.selected_object_id != 0

814:                 result.success = False

817:                     'snapshot_id and selected_object_id must be supplied '
818:                     'together'
819:                 )

823:                 return self._execute_selection(goal_handle, result)

825:                 goal_handle,
826:                 InspectionStage.WAITING_FOR_RGBD,
827:                 0,
828:                 0,
829:                 'Waiting for a new synchronized RGB-D snapshot',
830:             )

832:             messages = self._snapshot_buffer.wait_for_new(

834:                 self._snapshot_timeout_seconds,

839:                 depth_16u_scale_m=self._depth_16u_scale_m,

844:             inspection_began = time.monotonic()
845:             detections = self._pipeline.detect(
846:                 snapshot,
847:                 query,
848:                 progress=lambda stage, detections, objects, message: (
849:                     self._feedback(
850:                         goal_handle,
851:                         stage,
852:                         detections,
853:                         objects,
854:                         message,
855:                     )
856:                 ),
857:                 cancelled=lambda: goal_handle.is_cancel_requested,

877:                 item.distance_m for item in ranked_detections

893:             if goal_handle.is_cancel_requested:
894:                 raise InspectionFailure(FailureKind.CANCELLED, 'Observation canceled before tracking update')

896:                 session_id, snapshot_id, [d.label for d in detections],
897:                 [item.position for item in ranked_detections],

903:                     snapshot=snapshot,
904:                     detections=detections,

914:                 detections,
915:                 snapshot.stamp_ns,

919:                 tracking, self._target_frame,
920:             )
921:             result.success = True
922:             result.error_code = InspectScene.Result.ERROR_NONE

964:         cached = self._snapshot_cache.get(snapshot_id)
965:         if cached is None:

972:         if requested_session and (cached.tracking is None or cached.tracking.session_id != requested_session):
973:             result.error_code = InspectScene.Result.ERROR_INVALID_SELECTION

978:         if selected_id < 1 or selected_id > len(cached.detections):
979:             result.success = False
980:             result.error_code = InspectScene.Result.ERROR_INVALID_SELECTION

985:             goal_handle.abort()

992:             cached.capture_transform,
993:             progress=lambda stage, detections, objects, message: (
994:                 self._feedback(
995:                     goal_handle,
996:                     stage,
997:                     detections,
998:                     objects,
999:                     message,
1000:                 )
1001:             ),
1002:             cancelled=lambda: goal_handle.is_cancel_requested,
1003:             geometry_cache=cached.geometry_cache,
1004:         )
1005:         if cached.geometry_cache is not None:
1006:             self.get_logger().info(json.dumps({'event': 'inspection_geometry_cache',
1007:                 'snapshot_id': snapshot_id, 'object_id': selected_id,
1008:                 'hits': cached.geometry_cache.hits, 'misses': cached.geometry_cache.misses}))
1009:         detections_message = self._detections_message(
1010:             cached.detections,
1011:             cached.snapshot.stamp_ns,
1012:             cached.color_frame,

1187:         detections: Sequence[Detection2D],

1189:         frame_id: str,

1191:         detection_distances_m: Sequence[float | None],

1193:         representative_frame: str = 'base_link',

1196:             raise ValueError('Detection distances must match detections')

1199:         message.header.frame_id = frame_id
1200:         message.snapshot_id = snapshot_id
1201:         message.representative_frame = representative_frame
1202:         if tracking is not None:
1203:             message.tracking_session_id, message.tracking_epoch = tracking.session_id, tracking.epoch
1204:             message.missing_objects = [TrackedObjectReference(**vars(ref)) for ref in tracking.missing]

1206:             zip(detections, detection_distances_m, strict=True),

1212:             detected.detector_label = detection.detector_label
1213:             detected.confidence = detection.confidence
1214:             detected.sorting_category = detection.sorting_category
1215:             detected.distance_valid = distance_m is not None
1216:             detected.distance_m = 0.0 if distance_m is None else distance_m
1217:             if tracking is not None:
1218:                 tracked = tracking.detections[object_id - 1]
1219:                 detected.track_id, detected.tracking_state = tracked.track_id, tracked.state.value
1220:                 detected.position_valid = tracked.position is not None
1221:                 if tracked.position is not None:
1222:                     (detected.representative_position.x, detected.representative_position.y,
1223:                      detected.representative_position.z) = tracked.position
1224:             detected.median_depth_m = detection.median_depth_m
1225:             detected.estimated_width_m = detection.estimated_width_m
1226:             detected.estimated_height_m = detection.estimated_height_m
1227:             detected.x_min = detection.bbox.x_min
1228:             detected.y_min = detection.bbox.y_min
1229:             detected.x_max = detection.bbox.x_max
1230:             detected.y_max = detection.bbox.y_max
